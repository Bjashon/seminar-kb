\begin{frame}[label=poisson]{Алгебры Пуассона}
\small
\begin{defboxn}
\hlt{Алгебра Пуассона} — в.п. $V$ над $\F$ с единицей $1$, билинейными умножением $ab$ и скобкой $\{a,b\}$, такими что для всех $a,b,c$
\[ \renewcommand{\arraystretch}{1.3}\begin{array}{@{}l@{\hspace{0.7cm}}l@{}}
\ano{1.1}\begin{aligned}
ab&=ba\\
(ab)c&=a(bc)\\
1a&=a
\end{aligned}\quad\nm{умножение}
&
\ano{1.2}\begin{aligned}
\{a,b\}&=-\{b,a\}\\
\{a,\{b,c\}\}&=\{\{a,b\},c\}+\{b,\{a,c\}\}
\end{aligned}\quad\nm{скобка Ли}\\[4pt]
\multicolumn{2}{c}{\phantom{\nm{правило Лейбница}\qquad}\ano{1.3}\{a,bc\}=\{a,b\}\,c+b\,\{a,c\}\qquad\nm{правило Лейбница}}
\end{array} \]
\end{defboxn}
\begin{columns}[T,onlytextwidth]
\column{0.5\textwidth}
\only<2->{\begin{exbox}{Пример: $\F[p,q]$, \ \ftip[-6.3cm]{pq}{$\{f,g\}=\partial_pf\,\partial_qg-\partial_qf\,\partial_pg$}{Это скобка Пуассона: кососимметрична (f и g меняются местами); правило Лейбница — потому что ∂p и ∂q дифференцирования, ∂(gh) = ∂g·h + g·∂h; тождество Якоби — прямой счёт.}{Почему это скобка Пуассона:
\[ \begin{aligned}
\{g,f\}&=-\{f,g\}\\
\partial(gh)&=\partial g\,h+g\,\partial h &&\nm{$\partial=\partial_p,\ \partial_q$}\\
\{f,gh\}&=\{f,g\}\,h+g\,\{f,h\}
\end{aligned} \]
Тождество Якоби — прямой счёт: члены со вторыми производными сокращаются.}}
\[ \begin{aligned}
\{p,q\}&=\partial_pp\,\partial_qq-\partial_qp\,\partial_pq=1\cdot1-0\cdot0=1\\
\{p,q^2\}&=\{p,q\}\,q+q\,\{p,q\}=2q
\end{aligned} \]
\end{exbox}}
\column{0.46\textwidth}
\onslide<3->
\bul \hlt{Скобка с единицей равна нулю:}
\[ \begin{aligned}
\{a,1\}&=\{a,1\cdot1\}=2\{a,1\}\\
\{a,1\}&=0
\end{aligned} \]
\onslide<4->
\bul Скобка однозначно задаётся значениями на образующих; так на \ftip[-2.6cm]{sg}{$S(\mathfrak g)$}{S(g) — многочлены от базиса e₁, …, eₙ алгебры Ли g. \[ \begin{aligned}
S(\mathfrak g)&=\F[e_1,\dots,e_n]\\

\end{columns}

\begin{frame}[label=super]{Супералгебры}
\small
\begin{defboxn}
Пусть $\F$ — поле, $\operatorname{char}\F=0$. \hlt{Супералгебра} — в.п. $V$ над $\F$ с билинейным умножением $ab$ и \hlt{$\Z_2$-градуировкой} — разложением в прямую сумму, согласованным с умножением:
\[ V=V_0\oplus V_1,\qquad V_iV_j\subseteq V_{(i+j)\bmod 2}\quad (i,j\in\{0,1\}). \]
Элементы $V_0$ — \hlt{чётные}, $V_1$ — \hlt{нечётные}, те и другие — \hlt{однородные}; \hlt{чётность} $|a|=i$ при $a\in V_i$. Всякий $a$ однозначно раскладывается: $a=a_0+a_1$, $a_i\in V_i$.
\end{defboxn}
\vspace{-2pt}\noindent{\color{applegreen}\large$\Rightarrow$}\enspace Далее $V=V_0\oplus V_1$ — супералгебра, $a,b,c$ в формулах с чётностями — однородные.\par\vspace{4pt}
\only<2->{\begin{exbox}{Пример: матрицы $2\times2$; \ $V_0$ — диагональные, \ $V_1$ — внедиагональные}
\[ \begin{aligned}
E_{11}E_{12}&=E_{12}\in V_1 &&\nm{$E_{ij}$ — матричные единицы; \ $0+1=1$}\\
E_{12}E_{21}&=E_{11}\in V_0 &&\nm{$1+1=0$ по модулю 2}
\end{aligned} \]
\end{exbox}}
\end{frame}

\begin{frame}[label=supercomm]{Суперкоммутативные супералгебры}
\small
\begin{defboxn}
$V$ \hlt{суперкоммутативна}, если для всех $a,b$
\[ \ano{1.4}ab=(-1)^{|a||b|}\,ba\qquad\nm{правило Кошуля: $a\,b\to(-1)^{|a||b|}\,b\,a$} \]
Значит, чётный элемент коммутирует со всеми, а нечётные антикоммутируют; в частности, при $|a|=1$ по (1.4) $a^2=-a^2$, то есть $a^2=0$.
\end{defboxn}
\only<2->{\begin{exbox}{Пример: матрицы $2\times2$ — супералгебра, но не суперкоммутативная}
\[ \begin{aligned}
E_{12}E_{21}&=E_{11} &&\nm{$E_{12}$, $E_{21}$ нечётные}\\
-E_{21}E_{12}&=-E_{22}\neq E_{11} &&\nm{(1.4) не выполнено}
\end{aligned} \]
\end{exbox}}
\end{frame}

\begin{frame}[label=superder]{Супердифференцирования}
\small
\begin{defboxn}
\hlt{Супердифференцирование чётности $p\in\{0,1\}$} — линейное $d\colon V\to V$, сдвигающее чётность на $p$,
\[ d(V_i)\subseteq V_{(i+p)\bmod 2},\qquad\text{то есть}\qquad |d(a)|=(|a|+p)\bmod 2, \]
и удовлетворяющее правилу Лейбница со знаком:
\[ \ano{1.5}d(ab)=d(a)\,b+(-1)^{p|a|}\,a\,d(b)\qquad\nm{правило Кошуля: $d\,a\to(-1)^{p|a|}\,a\,d$} \]
При $p=0$ это обычное дифференцирование.
\end{defboxn}
\only<2->{\begin{exbox}{Пример: на $\F[x\mid\xi_1,\xi_2]$ производная $d=\partial_{\xi_1}$, \ $p=1$}
\[ \begin{aligned}
\partial_{\xi_1}(\xi_1\xi_2)&=\xi_2\\
\partial_{\xi_1}(\xi_2\xi_1)&=(\partial_{\xi_1}\xi_2)\,\xi_1+(-1)^{1\cdot1}\,\xi_2\,\partial_{\xi_1}\xi_1=-\xi_2 &&\nm{по \fn{1.5}}
\end{aligned} \]
\end{exbox}}
\end{frame}

\begin{frame}[label=freealg]{Свободная алгебра и её базис}
\small
\begin{defboxn}
Пусть $\mathcal K$ — класс алгебр, заданный тождествами (например, алгебры Ли), $X$ — множество (для супералгебр — с чётностями элементов).\par\smallskip
\hlt{Свободная алгебра} класса $\mathcal K$ на $X$ — алгебра $F(X)\in\mathcal K$, порождённая $X$, такая что для любой $A\in\mathcal K$ и любого отображения $\varphi$, сохраняющего чётность,
\[ \varphi\colon X\to A\quad\Longrightarrow\quad\exists!\ \text{гомоморфизм}\ \Phi\colon F(X)\to A,\qquad \Phi|_X=\varphi. \]
Элементы $F(X)$ — выражения от букв, равные только в силу тождеств класса.
\end{defboxn}
\only<2->{\begin{exbox}{Пример: $\mathcal K$ — ассоциативные коммутативные алгебры с $1$, \ $X=\{x,y\}$; \ $F(X)=\F[x,y]$}
\[ \begin{aligned}
\varphi\colon\ &x\mapsto t^2,\ \ y\mapsto t+1 &&\nm{любое отображение в $A=\F[t]$}\\
\Phi(f)&=f(t^2,\ t+1) &&\nm{гомоморфизм — подстановка}\\
\Phi(x^2y+3)&=t^4(t+1)+3 &&\nm{единствен: $\Phi(x^ay^b)=\varphi(x)^a\varphi(y)^b$}
\end{aligned} \]
\end{exbox}}
\end{frame}

\begin{frame}[label=good]{Алгебры Ли: хорошие слова Холла}
\small
\begin{defboxn}
Пусть $X$ — упорядоченное множество букв; \hlt{слова} — буквы и $[u,v]$ от слов $u,v$, \hlt{степень} — число букв. \hlt{Хорошие слова} и порядок на них — индукцией по степени:
\begin{itemize}\setlength{\itemsep}{0pt}
  \item буквы $x\in X$; порядок — как в $X$;
  \item слова $w=[u,v]$, где $u,v$ — хорошие и
  \[ u>v,\qquad u_2\le v\ \ \text{при}\ u=[u_1,u_2]. \]
\end{itemize}
Порядок: большая степень — больше; при равной $[u,v]>[u',v']\iff u>u'$ или $u=u'$, $v>v'$.
\end{defboxn}
\only<2->{\begin{exbox}{Пример: $x_1<x_2<x_3$, каждая буква по разу}
\[ \begin{aligned}
&[x_2,x_1]<[x_3,x_1]<[x_3,x_2] &&\nm{степень 2}\\
&[[x_2,x_1],x_3],\ \ [[x_3,x_1],x_2] &&\nm{хорошие степени 3}\\
&[[x_3,x_2],x_1]=[[x_3,x_1],x_2]-[[x_2,x_1],x_3] &&\nm{не хорошее: $x_2>x_1$; по Якоби}
\end{aligned} \]
\end{exbox}}
\onslide<3->
\textbf{Теорема М.~Холла (1950).} Хорошие слова — базис свободной алгебры Ли; мультилинейных на $k$ буквах — $(k-1)!$.
\end{frame}

\begin{frame}[label=superlie]{Супералгебры Ли: квадраты нечётных}
\small
Для нечётного $v$ суперкососимметричность при $a=b=v$ выполняется тождественно, и квадрат $[v,v]$ \hlt{может быть ненулевым}:
\[ [v,v]=-(-1)^{1\cdot1}[v,v]=[v,v]. \]
\only<2->{\begin{exbox}{Пример: $\mathfrak{gl}(1|1)$, \ $v=E_{12}+E_{21}$ нечётен; для нечётных $[a,b]=ab+ba$}
\[ [v,v]=2v^2=2(E_{12}E_{21}+E_{21}E_{12})=2(E_{11}+E_{22})\neq0 . \]
\end{exbox}}
\onslide<3->
По тождеству Якоби скобки, содержащие квадрат, выражаются через скобки без него; поэтому квадраты входят в базис только сами по себе:
\[ \begin{aligned}
{}[v,[v,v]]&=0\\
{}[[v,v],w]&=2\,[v,[v,w]]
\end{aligned} \]
\only<4->{\begin{thmbox}{Теорема (А.\,С.~Штерн, 1986)}
Базис свободной супералгебры Ли: хорошие слова и \hlt{квадраты $[v,v]$ нечётных хороших слов $v$}.
\end{thmbox}}
\end{frame}

\begin{frame}[label=shestakov]{Алгебры Пуассона: шаблон Шестакова}
\small
\begin{thmbox}{Теорема (И.\,П.~Шестаков, 1993)}
Свободная алгебра Пуассона на $X$ — это $S(\mathrm{Lie}(X))$: базис — \hlt{одночлены от хороших слов}.
\end{thmbox}
\only<2->{\begin{exbox}{Пример: мультилинейные элементы от $x_1<x_2<x_3$; \ всего $1+3+2=6=3!$}
\[ x_1x_2x_3;\qquad \{x_2,x_1\}x_3,\ \ \{x_3,x_1\}x_2,\ \ \{x_3,x_2\}x_1;\qquad \{\{x_2,x_1\},x_3\},\ \ \{\{x_3,x_1\},x_2\} \]
\end{exbox}}
\onslide<3->
\textbf{Схема доказательства} (её повторяет доказательство теоремы 12):
\begin{itemize}\setlength{\itemsep}{1pt}
\item \hlt{Порождение.} По правилу Лейбница скобка с произведением — сумма произведений скобок, а скобка хороших слов — комбинация хороших слов (Холл). Значит, любой элемент — комбинация одночленов от хороших слов:
\[ \{x_1,x_2^2\}=\{x_1,x_2\}x_2+x_2\{x_1,x_2\}=-2\,x_2\{x_2,x_1\}\qquad\nm{одночлен от хороших слов $x_2$, $\{x_2,x_1\}$} \]
\item \hlt{Независимость.} Гомоморфизм из свободной алгебры в модель $S(\mathrm{Lie}(X))$ (скобка Ли, продолженная по правилу Лейбница), тождественный на $X$, переводит эти одночлены в базис модели.
\end{itemize}
\end{frame}

\begin{frame}[label=shestakovn]{\ft{Алгебры Пуассона: почему $n!$}{Алгебры Пуассона: почему n!}}
\small
\textbf{Утверждение.} $\dim P_n=n!$, где $P_n$ — мультилинейная часть свободной алгебры Пуассона на $x_1,\dots,x_n$.\par\smallskip
\hlt{Базис $P_n$.} По теореме Шестакова это произведения $u_1\cdots u_r$ хороших слов, в которые каждая буква входит ровно один раз. Такое произведение задаётся разбиением $\{x_1,\dots,x_n\}=B_1\sqcup\dots\sqcup B_r$ (порядок блоков не важен: умножение коммутативно) и мультилинейным хорошим словом на каждом блоке $B_j$.\par\smallskip
\hlt{Счёт.} По теореме Холла мультилинейных хороших слов на $k$ буквах $(k-1)!$ — столько же, сколько циклических порядков на $k$-элементном множестве. Перестановка из $S_n$ однозначно задаётся разбиением на циклы и циклическим порядком на каждом цикле, поэтому
\[ \dim P_n=\sum_{\{B_1,\dots,B_r\}}\ \prod_{j=1}^{r}\bigl(|B_j|-1\bigr)!=|S_n|=n!. \]
\only<2->{\begin{exbox}{Пример: $n=3$ — шесть элементов с предыдущего слайда}
\[ \underbrace{(3-1)!}_{\{x_1,x_2,x_3\}}+\underbrace{3\cdot1!\cdot0!}_{\{x_i,x_j\}\sqcup\{x_k\}}+\underbrace{0!\cdot0!\cdot0!}_{\{x_1\}\sqcup\{x_2\}\sqcup\{x_3\}}=2+3+1=6=3! \]
\end{exbox}}
\end{frame}

\begin{frame}[label=setM]{\ft{Скобочная часть: множество $M$}{Скобочная часть: множество M}}
\small
\begin{defboxn}
Пусть $X=\{1<x_1<x_2<\cdots\}$ — буквы с заданными чётностями, $|1|=0$. Слова над $X$ строятся нечётной скобкой $\{\,,\}$, их чётность $|\{u,v\}|=|u|+|v|+1$; хорошие слова и их порядок — как у Холла. \hlt{Множество $M$} состоит из хороших слов и квадратов $\{v,v\}$ хороших слов чётности $0$:
\[ M=\bigl\{\,u,\ \{v,v\}\ \bigm|\ u,v\ \text{— хорошие слова над}\ X,\ |v|=0\,\bigr\}. \]
Квадраты сравниваются с остальными так же, как слова; по возрастанию $M=\{e_1=1<e_2<e_3<\cdots\}$.
\end{defboxn}
\only<2->{\begin{thmbox}{Лемма 4}
$M$ — базис свободной \ftip[-5.2cm]{oddlie}{нечётной супералгебры Ли}{кососимметричность (3) и Якоби (4): супералгебра Ли относительно чётности ‖a‖ = |a| + 1}{Супералгебра Ли относительно обращённой чётности $\rp a=|a|+1$:
\[ \begin{aligned}
\{a,b\}&=-(-1)^{\rp a\rp b}\{b,a\} &&\nm{3}\\
\{a,\{b,c\}\}&=\{\{a,b\},c\}+(-1)^{\rp a\rp b}\{b,\{a,c\}\} &&\nm{4}
\end{aligned} \]} $\LL$ на $X$. Квадраты — у \hlt{чётных} слов (теорема Штерна), так что $\{1,1\}\in M$.
\end{thmbox}}
\end{frame}

\end{defboxn}

$U$ — базис свободной унитальной обобщённой алгебры Герстенхабера $\G$ на $X$. Более того, $\G$ изоморфна суперполиномам $\Sm=S(\LL')$ с явно заданной скобкой.

\[ \{1,1\},\qquad \{1,1\}\,x_i,\qquad \{1,x_i\}. \]

n=2:&\quad \{x_1,1\}\,x_2,\quad \{x_2,1\}\,x_1,\quad \{\{x_1,1\},x_2\},\quad \{\{x_2,1\},x_1\} && 4=2\cdot2!
\end{aligned} \]

\begin{frame}[label=multi]{\ft{Мультилинейная часть: $n\cdot n!$}{Мультилинейная часть: n·n!}}
\small
\begin{defboxn}
Пусть $X=\{1,x_1,\dots,x_n\}$. \hlt{Мультилинейная часть} $\mathrm{GenGer}(n)$ — линейная оболочка тех элементов $U$, в которые каждая из букв $1,x_1,\dots,x_n$ входит ровно один раз; \hlt{буква $1$ тоже считается}.
\end{defboxn}
\only<2->{\begin{thmbox}{Теорема 14}
\centering\Large $\dim\mathrm{GenGer}(n)=n\cdot n!$
\end{thmbox}}
\onslide<3->
\vskip0.1cm
\centering\small\renewcommand{\arraystretch}{1.2}
\begin{tabular}{@{}lcccc@{}}
\toprule
$n$ & 1 & 2 & 3 & 4\\
\midrule
обобщённые алгебры Герстенхабера (и Пуассона, K17): $n\cdot n!$ & \textbf{1} & \textbf{4} & \textbf{18} & \textbf{96}\\
алгебры Пуассона и Герстенхабера, по $x_1,\dots,x_n$: $n!$ & 1 & 2 & 6 & 24\\
\bottomrule
\end{tabular}
\vskip0.25cm
\raggedright
{\small При $n=2$ — четыре элемента \go{cor13}{примера 2}. Почему $n\cdot n!$ — на следующем слайде.}
\end{frame}

\begin{frame}[label=perm]{Доказательство теоремы 14}
\small
\hlt{Базис.} Элементы $U$, в которые каждая из букв $1,x_1,\dots,x_n$ входит ровно один раз, — произведения хороших слов: квадрат $\{v,v\}$ содержит каждую букву $v$ дважды. Такое произведение задаётся разбиением $\{1,x_1,\dots,x_n\}=B_1\sqcup\dots\sqcup B_r$ и хорошим словом на каждом блоке; \hlt{блока $\{1\}$ нет}, потому что множителя $1$ в $U$ нет.\par\smallskip
\onslide<2->
\hlt{Счёт.} Как для \go{shestakovn}{алгебр Пуассона}: хороших слов на блоке из $k$ букв $(k-1)!$, столько же, сколько циклических порядков, а запрет блока $\{1\}$ — запрет неподвижной точки $1$:
\[ \dim\mathrm{GenGer}(n)=\#\{\sigma\in S_{n+1}\mid\sigma(1)\neq1\}=(n+1)!-n!=n\cdot n!. \]
\only<3->{\begin{exbox}{Пример: $n=2$, буквы $1,x_1,x_2$, \ $3!=6$ перестановок}
\small\renewcommand{\arraystretch}{1.15}
\begin{tabular}{@{}l@{\qquad}l@{}}
$(1\,x_1)(x_2)$, \ $(1\,x_2)(x_1)$ & $\{x_1,1\}\,x_2$, \ $\{x_2,1\}\,x_1$\\
$(1\,x_1\,x_2)$, \ $(1\,x_2\,x_1)$ & $\{\{x_1,1\},x_2\}$, \ $\{\{x_2,1\},x_1\}$\\
{\color{mute}$(1)(x_1)(x_2)$, \ $(1)(x_1\,x_2)$} & {\color{mute}$1$ неподвижна — не считаются}\\
\end{tabular}\\[0.3em]
Итого $6-2=4=2\cdot2!$.
\end{exbox}}
\end{frame}

\begin{frame}[label=kantor]{Откуда поправка: йордановы скобки и двойник Кантора}
\begin{defbox}{Двойник Кантора $K(A)=A\oplus Ax$}
$A$ — суперкоммутативная супералгебра со скобкой $\{\,,\}$, $Ax$ — копия $A$; умножение:
$$a\ast b=ab,\qquad a\ast bx=(ab)x,\qquad ax\ast b=(-1)^{|b|}(ab)x,\qquad ax\ast bx=(-1)^{|b|}\{a,b\}.$$
\end{defbox}
\begin{itemize}\small\setlength{\itemsep}{0.35em}
  \item Кантор (1990): для скобки Пуассона $K(A)$ — \ftip[-4.4cm]{jordan}{йорданова супералгебра}{суперкоммутативная алгебра с тождеством Йордана (x²y)x = x²(yx), с суперзнаками}{Суперкоммутативная супералгебра с тождеством Йордана; для чётных $x,y$
\[ (x^2y)\,x=x^2\,(yx), \]
для остальных — со знаками по правилу знаков.}.
  \item Кинг--МакКриммон (1992): $K(A)$ йорданова $\iff$ $\{\,,\}$ — \hlt{йорданова скобка}: Лейбниц и Якоби выполнены с поправками через $\{a,1\}$.
  \item Замена $\{a,b\}\mapsto\{a,b\}+\tfrac12\bigl(a\{b,1\}-\{a,1\}b\bigr)$ делает йорданову скобку обобщённой пуассоновой: Якоби точное, поправка только в Лейбнице.
  \item Нечётный вариант — \hlt{обобщённые алгебры Герстенхабера} (Кантарини--Кац 2010).
\end{itemize}
\end{frame}

\begin{frame}[label=po]{\ft{Примеры с $\D\neq0$: $PO(n,n+1)$}{Примеры с ненулевым D: PO(n,n+1)}}
\small
К $PO(n,n)$ добавим нечётную переменную $\tau$: $PO(n,n+1)=\F[x_1,\dots,x_n\mid\xi_1,\dots,\xi_n,\tau]$ со скобкой ($\{\,,\}_H$ — скобка Баттина, $E$ — оператор Эйлера)
$$\{f,g\}_K=\{f,g\}_H+(E-2)(f)\,\frac{\partial g}{\partial\tau}+(-1)^{|f|}\frac{\partial f}{\partial\tau}\,(E-2)(g),\qquad
E=\sum_{i=1}^n\Bigl(x_i\frac{\partial}{\partial x_i}+\xi_i\frac{\partial}{\partial\xi_i}\Bigr).$$
\begin{itemize}\setlength{\itemsep}{0.15em}
  \item $\D(g)=\{1,g\}_K=(E-2)(1)\,\partial g/\partial\tau=-2\,\partial g/\partial\tau\neq0$; \ например, $\{1,\tau\}_K=-2$, но $\{1,1\}_K=0$.
  \item Здесь $\D$ — всё-таки дифференцирование, так как $\{1,1\}=0$. В свободной алгебре будет иначе.
\end{itemize}
\normalsize
\begin{thmbox}{Теорема (Кантарини--Кац, 2010)}
Всякая простая линейно компактная обобщённая алгебра Герстенхабера калибровочно эквивалентна $PO(n,n)$ или $PO(n,n+1)$.
\end{thmbox}
\end{frame}

\begin{frame}[label=jacobi]{Якобиатор}
\begin{thmbox}{Лемма 9}
Пусть выполнены \ax{1}, \ax{2}, \ax{3}, \ax{5} (без тождества Якоби), и
$$J(a,b,c)=\{a,\{b,c\}\}-\{\{a,b\},c\}-(-1)^{\rp a\rp b}\{b,\{a,c\}\}.$$
Тогда
\begin{enumerate}
  \item[(i)] $J(a,b,c)=-(-1)^{\rp a\rp b}J(b,a,c)=-(-1)^{\rp b\rp c}J(a,c,b)$;
  \item[(ii)] $J(a,b,xy)=J(a,b,x)\,y+(-1)^{(|a|+|b|)|x|}\,x\,J(a,b,y)\;{\color{deepgreen}-\,J(a,b,1)\,xy}$.
\end{enumerate}
\end{thmbox}
\begin{itemize}\small
  \item \tip{Якобиатор}{мера нарушения тождества Якоби: J = 0 на всех тройках ⇔ тождество Якоби выполнено} — «дифференцирование с поправкой»; без поправки это классический факт для алгебр Пуассона.
  \item Доказательство: раскрыть три члена по \ax{5}, лишние члены сокращаются попарно.
  \item Общая формула, может быть полезна сама по себе.
\end{itemize}
\end{frame}

\begin{frame}[label=finish]{Тождество Якоби и завершение}
