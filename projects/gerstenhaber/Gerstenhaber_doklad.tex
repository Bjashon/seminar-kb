\partslide{01}{Контекст}

\begin{frame}{Скобка Пуассона}
\small
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Механика} (Пуассон, 1809). На функциях от $p_i,q_i$:
$$\{f,g\}=\sum_{i=1}^n\Bigl(\frac{\partial f}{\partial p_i}\frac{\partial g}{\partial q_i}-\frac{\partial f}{\partial q_i}\frac{\partial g}{\partial p_i}\Bigr);$$
эволюция наблюдаемой при гамильтониане $H$: $\dot f=\{H,f\}$.
\end{column}
\begin{column}{0.49\textwidth}
\textbf{Скобка Ли--Пуассона} на $S(\mathfrak g)=\F[\mathfrak g]$:
$$\{a_1\cdots a_m,\,b_1\cdots b_n\}=\sum_{i,j}[a_i,b_j]\,a_{\hat\imath}\,b_{\hat\jmath},$$
$a_{\hat\imath}$ — произведение без $a_i$: «каждый множитель с каждым».
\end{column}
\end{columns}
\normalsize
\begin{defbox}{Алгебра Пуассона}
Коммутативная ассоциативная алгебра с $1$ и скобкой Ли $\{\,,\}$, для которой $\{a,bc\}=\{a,b\}c+b\{a,c\}$ — правило Лейбница: $\{a,\cdot\}$ — дифференцирование.
\end{defbox}
{\small Пуассоновы многообразия, квантование, квантовые группы, операды; базис свободной алгебры Пуассона — Шестаков 1993.}
\end{frame}

\begin{frame}{Супералгебры и правило знаков}
\begin{itemize}\setlength{\itemsep}{0.45em}
  \item $\F$ — поле характеристики $0$. Всё $\Z_2$-градуировано: $V=V_0\oplus V_1$, \ $|a|\in\Z_2$ — чётность однородного $a$.
  \item \hl{Правило знаков:} перестановка двух нечётных соседей даёт $-1$. Суперкоммутативность $ab=(-1)^{|a||b|}ba$; в частности $a^2=0$ для нечётного $a$.
  \item \hl{Супердифференцирование} чётности $p$: \ $d(ab)=d(a)\,b+(-1)^{p|a|}\,a\,d(b)$.
\end{itemize}
\begin{exbox}{Суперполиномы $S(V)=\F[x_1,\dots,x_n\mid\xi_1,\dots,\xi_m]$}
$x_i$ чётные, $\xi_j$ нечётные: $\xi_j\xi_k=-\xi_k\xi_j$, $\xi_j^2=0$. Базис — одночлены $x^\alpha\xi_{j_1}\cdots\xi_{j_r}$, $j_1<\dots<j_r$.
Производные $\partial/\partial x_i$ (чётные) и $\partial/\partial\xi_j$ (нечётные) — супердифференцирования:
$$\frac{\partial}{\partial\xi_1}(\xi_2\xi_1)=-\frac{\partial}{\partial\xi_1}(\xi_1\xi_2)=-\xi_2 .$$
\end{exbox}
{\small Супералгебра Ли: $[a,b]=-(-1)^{|a||b|}[b,a]$, \quad $[a,[b,c]]=[[a,b],c]+(-1)^{|a||b|}[b,[a,c]]$.}
\end{frame}

\begin{frame}{Нечётная скобка и обращённая чётность}
\begin{itemize}\setlength{\itemsep}{0.2em}
  \item Скобка \hl{нечётна}: $|\{a,b\}|=|a|+|b|+1$. Какие знаки ставить в кососимметричности и Якоби?
  \item \hl{Обращённая чётность} $\rp a=|a|+1$; \ $\Pi V$ — то же пространство с чётностью $\rp{\cdot}$. Тогда $\rp{\{a,b\}}=\rp a+\rp b$: в $\Pi V$ скобка чётная.
\end{itemize}
\begin{defbox}{Нечётная супералгебра Ли}
$$\{a,b\}=-(-1)^{\rp a\rp b}\{b,a\},\qquad
  \{a,\{b,c\}\}=\{\{a,b\},c\}+(-1)^{\rp a\rp b}\{b,\{a,c\}\}.$$
Это ровно супералгебра Ли на $\Pi V$. Свободная нечётная супералгебра Ли на $X$ — свободная супералгебра Ли на $X$ с обращёнными чётностями.
\end{defbox}
{\small\textbf{Знаки.} Для чётных $a,b$: $\rp a\rp b=1$, $\{a,b\}=+\{b,a\}$ — скобка \hl{симметрична}, и $\{a,a\}$ ничем не ограничен. Для нечётного $v$: $\{v,v\}=-\{v,v\}=0$.}
\end{frame}

\begin{frame}{Алгебры Герстенхабера}
\begin{defbox}{Алгебра Герстенхабера (нечётная супералгебра Пуассона)}
Ассоциативная суперкоммутативная супералгебра с $1$ и нечётной скобкой, для которой $\Pi V$ — супералгебра Ли и
$$\{a,bc\}=\{a,b\}c+(-1)^{\rp a|b|}\,b\{a,c\},$$
$\{a,\cdot\}$ — супердифференцирование чётности $\rp a$; отсюда $\{a,1\}=\{1,a\}=0$.
\end{defbox}
\begin{itemize}\small\setlength{\itemsep}{0.15em}
  \item \textbf{Откуда:} когомологии Хохшильда $HH^\bullet(A,A)$ — скобка Герстенхабера (1963), отсюда название;
  \item поливекторные поля со скобкой Схоутена; $\Lambda(\mathfrak g)$ для алгебры Ли $\mathfrak g$;
  \item формализм Баталина--Вилковыского в физике: антискобка, BV-алгебры.
\end{itemize}
\end{frame}

\begin{frame}{Два примера}
\begin{exbox}{$PO(n,n)$ — поливекторные поля: $\F[x_1,\dots,x_n\mid\xi_1,\dots,\xi_n]$, \ $\xi_i\leftrightarrow\partial/\partial x_i$}
\small Скобка Баттина ($\{x_i,\xi_j\}=\delta_{ij}$, остальные скобки букв — нули):
$$\{f,g\}=\sum_{i=1}^n\Bigl(\frac{\partial f}{\partial x_i}\frac{\partial g}{\partial\xi_i}+(-1)^{|f|}\frac{\partial f}{\partial\xi_i}\frac{\partial g}{\partial x_i}\Bigr).$$
\end{exbox}
\begin{exbox}{$\Lambda(\mathfrak g)$ со скобкой Схоутена}
\small Внешняя алгебра алгебры Ли $\mathfrak g$, элементы $\mathfrak g$ нечётные; $\{a,b\}=[a,b]$ на $\mathfrak g$, $\{1,\cdot\}=0$, дальше по Лейбницу. Для $\mathfrak g=\langle h,e\rangle$, $[h,e]=2e$ ($\rp h=\rp e=0$):
$$\{h,he\}=\{h,h\}e+h\{h,e\}=0+2he,\qquad \{e,he\}=\{e,h\}e+h\{e,e\}=-2e^2+0=0 .$$
\end{exbox}
\end{frame}

\begin{frame}{Обобщённые супералгебры Пуассона}
\begin{defbox}{Правило Лейбница с поправкой (чётная скобка)}
$$\{a,bc\}=\{a,b\}c+(-1)^{|a||b|}\,b\{a,c\}-\D(a)\,bc,\qquad \D(a)=\{a,1\}.$$
$\{a,\cdot\}$ — дифференцирование плюс умножение на элемент (оператор первого порядка). Здесь $\{1,1\}=-\{1,1\}=0$, и $\D$ — дифференцирование.
\end{defbox}
\begin{exbox}{Пример}
\small $\F[x]$, $\{f,g\}=fg'-f'g$ — скобка векторных полей $f\,d/dx$ (Якоби есть); $\{f,1\}=-f'$ и
$$\{f,gh\}=fg'h+fgh'-f'gh=\{f,g\}h+g\{f,h\}-\{f,1\}\,gh .$$
\end{exbox}
{\small Простые линейно компактные: Кантарини--Кац 2007; свободные: Кайгородов 2017.}
\end{frame}

\begin{frame}{Откуда поправка: йордановы скобки и двойник Кантора}
\begin{defbox}{Двойник Кантора $K(A)=A\oplus Ax$}
$A$ — суперкоммутативная супералгебра со скобкой $\{\,,\}$, $Ax$ — копия $A$; умножение:
$$a\ast b=ab,\qquad a\ast bx=(ab)x,\qquad ax\ast b=(-1)^{|b|}(ab)x,\qquad ax\ast bx=(-1)^{|b|}\{a,b\}.$$
\end{defbox}
\begin{itemize}\small\setlength{\itemsep}{0.35em}
  \item Кантор (1990): для скобки Пуассона $K(A)$ — йорданова супералгебра.
  \item Кинг--МакКриммон (1992): $K(A)$ йорданова $\iff$ $\{\,,\}$ — \hl{йорданова скобка}: Лейбниц и Якоби выполнены с поправками через $\{a,1\}$.
  \item Замена $\{a,b\}\mapsto\{a,b\}+\tfrac12\bigl(a\{b,1\}-\{a,1\}b\bigr)$ делает йорданову скобку обобщённой пуассоновой: Якоби точное, поправка только в Лейбнице (Кантарини--Кац 2007, Кайгородов 2017).
  \item Нечётный вариант — \hl{обобщённые алгебры Герстенхабера} (Кантарини--Кац 2010).
\end{itemize}
\end{frame}

\begin{frame}{Вопрос и ответ}
\begin{thmbox}{Вопрос}
$X=\{1<x_1<x_2<\cdots\}$ — буквы с чётностями, $|1|=0$. Найти базис свободной унитальной обобщённой алгебры Герстенхабера $\G$ на $X$ и размерность её мультилинейной части.
\end{thmbox}
\begin{thmbox}{Ответ (Кайгородов--Вяткина)}
\begin{itemize}\setlength{\itemsep}{0.2em}
  \item Базис — суперкоммутативные одночлены от \hl{хороших слов Холла} и квадратов $\{v,v\}$ чётных хороших слов, где $1$ — \hl{буква скобки}.
  \item Новое явление: $\{1,1\}\neq0$ в $\G$, поэтому $\D=\{1,\cdot\}$ — не дифференцирование.
  \item $\dim\mathrm{GenGer}(n)=n\cdot n!$ — как в чётном случае (Кайгородов 2017).
\end{itemize}
\end{thmbox}
\end{frame}

\partslide{02}{Обобщённые алгебры\\ Герстенхабера}

\begin{frame}{Определение}
\begin{defbox}{Определение 1}
$\Z_2$-градуированное пространство $V$ с $1\in V_0$ и операциями $ab$, $\{a,b\}$:
$$\begin{aligned}
&(1)\quad |ab|=|a|+|b|,\qquad |\{a,b\}|=|a|+|b|+1;\\
&(2)\quad (ab)c=a(bc),\qquad ab=(-1)^{|a||b|}ba,\qquad 1a=a;\\
&(3)\quad \{a,b\}=-(-1)^{\rp a\rp b}\{b,a\};\\
&(4)\quad \{a,\{b,c\}\}=\{\{a,b\},c\}+(-1)^{\rp a\rp b}\{b,\{a,c\}\};\\
&(5)\quad \{a,bc\}=\{a,b\}c+(-1)^{\rp a|b|}b\{a,c\}+(-1)^{\rp a}\D(a)\,bc,\qquad \D(a)=\{1,a\}.
\end{aligned}$$
\end{defbox}
\begin{itemize}\small
  \item $\D=0$ $\Rightarrow$ алгебры Герстенхабера.
  \item Это в точности \emph{odd generalized Poisson superalgebras} Кантарини--Каца (2010).
  \item Поправка в (5) — та же $-\{a,1\}bc$: по (3) $\ (-1)^{\rp a}\D(a)=-\{a,1\}$.
\end{itemize}
\end{frame}

\begin{frame}{Что меняет поправка}
\begin{itemize}\setlength{\itemsep}{0.35em}
  \item {\color{mute}Зам. 2.} \ $(1),(3),(4)$ $\iff$ $(\Pi V,\{\,,\})$ — супералгебра Ли.
  \item {\color{mute}Зам. 3.} \ $(5)$ $\iff$ $d_a(b)=\{a,b\}+(-1)^{\rp a}\D(a)\,b$ — супердифференцирование чётности $\rp a$.
  \item Подстановка $a=1$ в $(5)$:
  $$\D(bc)=\D(b)\,c+(-1)^{|b|}\,b\,\D(c)-\D(1)\,bc .$$
\end{itemize}
\begin{thmbox}{Вывод}
$\D$ — нечётное дифференцирование $\iff$ $\D(1)=\{1,1\}=0$; в общем случае дифференцирование — $\D-\D(1)\,\mathrm{id}$.\\[0.3em]
Но $(3)$ при $a=b=1$ даёт $\{1,1\}=\{1,1\}$: \ $\rp1=1$, условия \hl{нет}.
\end{thmbox}
\end{frame}

\begin{frame}{\ft{Пример: $\{1,1\}\neq0$}{Пример: {1,1} не нуль}}
\begin{exbox}{Пример 1}
$V=\F1\oplus\F\varepsilon$, \ $|\varepsilon|=1$, \ $\varepsilon^2=0$;\qquad
$\{1,1\}=\varepsilon,\quad \{1,\varepsilon\}=\{\varepsilon,1\}=\{\varepsilon,\varepsilon\}=0 .$
\end{exbox}
\begin{itemize}\setlength{\itemsep}{0.5em}
  \item $(3)$ при $a=b=1$: \ $\{1,1\}=\{1,1\}$.
  \item $(4)$: все скобки лежат в $\F\varepsilon$, а $\{\varepsilon,V\}=0$.
  \item $(5)$ при $a=b=c=1$: \quad $\varepsilon=\varepsilon+\varepsilon-\varepsilon$; \ остальные тройки дают $0=0$.
\end{itemize}
\vskip0.2cm
\begin{thmbox}{Итог}
$V$ — обобщённая алгебра Герстенхабера, но $\D$ не дифференцирование:
$$\D(1\cdot1)=\varepsilon\ \neq\ 2\varepsilon=\D(1)\cdot1+1\cdot\D(1).$$
\end{thmbox}
\end{frame}

\begin{frame}{\ft{Примеры с $\D\neq0$: $PO(n,n+1)$}{Примеры с ненулевым D: PO(n,n+1)}}
\small
К $PO(n,n)$ добавим нечётную переменную $\tau$: $PO(n,n+1)=\F[x_1,\dots,x_n\mid\xi_1,\dots,\xi_n,\tau]$ со скобкой ($\{\,,\}_H$ — скобка Баттина, $E$ — оператор Эйлера)
$$\{f,g\}_K=\{f,g\}_H+(E-2)(f)\,\frac{\partial g}{\partial\tau}+(-1)^{|f|}\frac{\partial f}{\partial\tau}\,(E-2)(g),\qquad
E=\sum_{i=1}^n\Bigl(x_i\frac{\partial}{\partial x_i}+\xi_i\frac{\partial}{\partial\xi_i}\Bigr).$$
\begin{itemize}\setlength{\itemsep}{0.15em}
  \item $\D(g)=\{1,g\}_K=(E-2)(1)\,\partial g/\partial\tau=-2\,\partial g/\partial\tau\neq0$; \ например, $\{1,\tau\}_K=-2$, но $\{1,1\}_K=0$.
  \item Здесь $\D$ — всё-таки дифференцирование, так как $\{1,1\}=0$. В свободной алгебре будет иначе.
\end{itemize}
\normalsize
\begin{thmbox}{Теорема (Кантарини--Кац, 2010)}
Всякая простая линейно компактная обобщённая алгебра Герстенхабера калибровочно эквивалентна $PO(n,n)$ или $PO(n,n+1)$.
\end{thmbox}
\end{frame}

% ================================================================ 3. базис
\partslide{03}{Свободная алгебра\\ и её базис}

\begin{frame}{Свободная обобщённая алгебра Герстенхабера}
\begin{itemize}\setlength{\itemsep}{0.55em}
  \item $X=\{1<x_1<x_2<\cdots\}$, \ $|1|=0$, \ чётности $|x_i|$ произвольны.
  \item $D_X$ — все неассоциативные слова от $X$ с операциями $\cdot$ и $\{\,,\}$.
  \item $I$ — идеал, порождённый тождествами $(1)$--$(5)$; \quad $\G=D_X/I$.
  \item Универсальность: любое сохраняющее чётность $\{x_i\}\to V$ однозначно продолжается до гомоморфизма $\G\to V$, $1\mapsto1$.
\end{itemize}
\vskip0.3cm
\begin{thmbox}{Ключевое наблюдение}
\raggedright
$1$ — одновременно \hl{единица умножения} и \hl{буква скобки}:
$\{1,x_i\}$, $\{1,1\}$, $\{\{x_i,1\},1\},\ \dots$ — новые элементы. Скобочная часть «не знает», что $1$ — единица; связь только через правило Лейбница.
\end{thmbox}
\end{frame}

\begin{frame}{Хорошие слова (базисные коммутаторы Холла)}
\begin{defbox}{Определение}
Буквы — хорошие слова степени 1. Слово $w$ степени $n>1$ \hl{хорошее}, если
\begin{enumerate}\small
  \item $w=\{u,v\}$, где $u,v$ — хорошие и $u>v$;
  \item если $u=\{u_1,u_2\}$, то $u_2\le v$.
\end{enumerate}
Порядок: по степени, при равной степени — лексикографически по $(u,v)$.
\end{defbox}
\begin{exbox}{Над $\{1<x_1<x_2\}$}
\raggedright
Хорошие: \ $\{x_1,1\},\ \{x_2,1\},\ \{x_2,x_1\},\quad \{\{x_1,1\},1\},\ \{\{x_1,1\},x_2\},\ \{\{x_2,1\},x_1\},\ \dots$\par\smallskip
Не хорошее: $\{\{x_2,x_1\},1\}$ — нарушено условие 2, так как $x_1>1$. По Якоби оно выражается через $\{\{x_2,1\},x_1\}$ и $\{\{x_1,1\},x_2\}$.
\end{exbox}
\end{frame}

\begin{frame}{Теорема Холла и супер-случай}
\begin{thmbox}{Теорема (М.~Холл 1950; Ширшов 1958)}
Хорошие слова — базис свободной алгебры Ли $\mathrm{Lie}(X)$; мультилинейных слов от $x_1,\dots,x_k$ ровно $(k-1)!$.
\end{thmbox}
\begin{thmbox}{Свободные супералгебры Ли (Штерн 1986; БМПЗ; БКЛМ)}
Базис: хорошие слова \hl{и квадраты $[v,v]$ нечётных хороших слов $v$}.
\end{thmbox}
{\small Почему квадраты: для нечётного $v$ бывает $[v,v]=2v^2\neq0$ (в $\mathfrak{gl}(1|1)$: $v=E_{12}+E_{21}$, $v^2=E_{11}+E_{22}$), а $[v,[v,v]]=0$ всегда по Якоби.}
\end{frame}

\begin{frame}{\ft{Множество $M$}{Множество M}}
$$M=\bigl\{\,u,\ \{v,v\}\ \bigm|\ u,v\ \text{— хорошие слова},\ |v|=0\,\bigr\},\qquad e_1=1<e_2<e_3<\cdots$$
\begin{thmbox}{Лемма 4}
$M$ — базис свободной нечётной супералгебры Ли $\LL$, порождённой $X$.
\end{thmbox}
\begin{itemize}\setlength{\itemsep}{0.45em}
  \item По замечанию 2: $\LL$ — свободная супералгебра Ли на $X$ с чётностью $\rp x$.
  \item Её базис: хорошие слова и квадраты слов с $\rp v=1$.
  \item $\rp v=1\iff|v|=0$: квадраты \hl{чётных} слов. В частности, $\{1,1\}\in M$ — квадрат чётного хорошего слова $1$.
\end{itemize}
\end{frame}

\begin{frame}{Главная теорема}
$$U=\bigl\{\,e_{i_1}^{k_1}\cdots e_{i_r}^{k_r}\ \bigm|\ r\ge0,\ 1<i_1<\dots<i_r,\ k_j\ge1,\ k_j=1\ \text{при}\ |e_{i_j}|=1\,\bigr\}$$
\centerline{\small\color{mute}суперкоммутативные одночлены от $M\setminus\{1\}$}
\vskip0.25cm
\begin{thmbox}{Теорема 12}
$U$ — базис свободной унитальной обобщённой алгебры Герстенхабера $\G$, порождённой $X$.
Более того, $\G\cong\Sm$ — свободная суперкоммутативная супералгебра над $\LL'=\spn(M\setminus\{1\})$
с явно заданной скобкой.
\end{thmbox}
\begin{itemize}\small
  \item Шаблон тот же, что у Шестакова: «свободная $=$ симметрическая алгебра над свободной скобочной частью».
  \item Новое: буква $1$ в скобках и базисный элемент $\{1,1\}$.
\end{itemize}
\end{frame}

\begin{frame}{Следствие и пример}
\begin{thmbox}{Следствие 13}
В $\G$: \ $\{1,1\}\neq0$, и $\D$ не является дифференцированием.
Более того, $\{1,1\}$, \ $\{1,1\}x_i$, \ $\{1,x_i\}$ \ линейно независимы.
\end{thmbox}
{\small\color{mute}Без теоремы 12: гомоморфизм $\G\to V$ (пример 1), $1\mapsto1$, $x_i\mapsto0$, переводит $\{1,1\}\mapsto\varepsilon\neq0$.}
\vskip0.3cm
\begin{exbox}{Пример 2: мультилинейные элементы $U$ по $1,x_1,\dots,x_n$}
$n=1$: \quad $\{x_1,1\}$\\[0.3em]
$n=2$: \quad $\{x_1,1\}\,x_2,\quad \{x_2,1\}\,x_1,\quad \{\{x_1,1\},x_2\},\quad \{\{x_2,1\},x_1\}$ \qquad ($4=2\cdot2!$)
\end{exbox}
\end{frame}

\begin{frame}{\ft{Лемма 5: $U$ порождает $\G$}{Лемма 5: U порождает G}}
\begin{itemize}\setlength{\itemsep}{0.45em}
  \item $\psi\colon\LL\to\G$ — гомоморфизм нечётных супералгебр Ли; $\psi(\LL)=\spn M$ (лемма 4).
  \item $\{u,v\}\in\spn U$ для $u,v\in U$ — индукция по $N=\deg_e u+\deg_e v$:
  $$\{u,v_1v_2\}=\{u,v_1\}v_2+(-1)^{\rp u|v_1|}v_1\{u,v_2\}+(-1)^{\rp u}\{1,u\}\,v_1v_2 .$$
  \item $\spn U$ замкнуто относительно обеих операций и содержит $X$ $\Rightarrow$ $\spn U=\G$.
\end{itemize}
\vskip0.2cm
\begin{exbox}{Почему не по числу букв}
$1a=a$ уменьшает число букв, а поправка в $(5)$ увеличивает:
$\{x_1,x_2x_3\}$ (4 буквы) содержит $(-1)^{\rp{x_1}}\{1,x_1\}x_2x_3$ (5 букв).
\end{exbox}
\end{frame}

\begin{frame}{Модельная алгебра}
\small
$\LL'=\spn(M\setminus\{1\})$, \ $\Sm=S(\LL')$, \ $\Lh=\F1\oplus\LL'\cong\LL$ (скобка переносится); \
$\D(u)=\{1,u\}$ для $u\in\Lh$.\\[0.2em]
Для $a=a_1\cdots a_m$: \ $a_{<i}=a_1\cdots a_{i-1}$, \ $a_{>i}=a_{i+1}\cdots a_m$, \ $a_{\hat\imath}=a_{<i}a_{>i}$.
\begin{defbox}{Формулы (7), (8)}
$$\D(a)=\sum_{i=1}^m(-1)^{|a_{<i}|}\,a_{<i}\{1,a_i\}a_{>i}\;-\;(m-1)\{1,1\}\,a,$$
$$\begin{aligned}
\{a,b\}^*=\sum_{i,j}(-1)^{|a_i||a_{>i}|+|b_j||b_{<j}|}\,a_{\hat\imath}\{a_i,b_j\}b_{\hat\jmath}
&-(m-1)\,a\D(b)+(n-1)(-1)^{\rp a}\D(a)\,b\\
&-(m-1)(n-1)\,a\D(1)\,b .
\end{aligned}$$
\end{defbox}
\begin{itemize}
  \item Двойная сумма: «$a_i$ — в правый конец $a$, $b_j$ — в левый конец $b$, скобка соседей».
  \item Остальное — поправки с $\D$; при $\D=0$ остаётся обычное бидифференцирование.
\end{itemize}
\end{frame}

\begin{frame}{Как читать формулу (8)}
\begin{itemize}\small\setlength{\itemsep}{0.3em}
  \item \hl{Двойная сумма} — «каждый множитель $a$ с каждым множителем $b$», знак — правило Кошуля за передвижения. При $\D=0$ на этом всё и кончается, как для $S(\mathfrak g)$ и $\Lambda(\mathfrak g)$.
  \item \hl{$(n-1)(-1)^{\rp a}\D(a)\,b$} — след поправки: каждое применение (5) добавляет копию $-\{a,1\}b$, а применений $n-1$.
  \item \hl{$-(m-1)\,a\D(b)$} — то же по первому аргументу, через кососимметричность.
  \item \hl{$-(m-1)(n-1)\,a\D(1)\,b$} — «поправка к поправке», когда произведения с обеих сторон. В чётном случае (K17) этого члена нет: там $\D(1)=\{1,1\}=0$.
\end{itemize}
\begin{exbox}{Откуда $n-1$: раскрываем $\{a,b_1b_2b_3\}$, $a\in\Lh$, по (5)}
\small
$$\begin{aligned}
\{a,b_1b_2b_3\}&=\{a,b_1\}\,b_2b_3\pm b_1\{a,b_2b_3\}-\{a,1\}\,b_1b_2b_3\\
&=\{a,b_1\}b_2b_3\pm b_1\{a,b_2\}b_3\pm b_1b_2\{a,b_3\}-2\,\{a,1\}\,b_1b_2b_3 .
\end{aligned}$$
Знаки $\pm$ — по правилу Кошуля; \ $-\{a,1\}=(-1)^{\rp a}\D(a)$.
\end{exbox}
\end{frame}

\begin{frame}{\ft{Свойства скобки $\{\,,\}^*$}{Свойства модельной скобки}}
\begin{thmbox}{Лемма 6}
(7), (8) не зависят от разложения на множители из $\Lh$, и
\begin{enumerate}\small
  \item[(i)] $\{u,v\}^*=\{u,v\}$, \ $\D(u)=\{1,u\}$ при $u,v\in\Lh$;\qquad
  (ii) $\{1,a\}^*=\D(a)$;
  \item[(iii)] $\D(bc)=\D(b)c+(-1)^{|b|}b\D(c)-\D(1)bc$;\qquad
  (iv) $|\{a,b\}^*|=|a|+|b|+1$.
\end{enumerate}
\end{thmbox}
\vskip0.1cm
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{thmbox}{Лемма 7}
$\{a,b\}^*=-(-1)^{\rp a\rp b}\{b,a\}^*$
\end{thmbox}
\end{column}
\begin{column}{0.48\textwidth}
\begin{thmbox}{Лемма 8}
Правило $(5)$ для $\{\,,\}^*$
\end{thmbox}
\end{column}
\end{columns}
\vskip0.1cm
{\small Проверки прямые: формула (8) — замкнутая, без рекурсии. Корректность (лемма 6) нужна, потому что множители можно переставлять и вставлять лишние единицы: $1\in\Lh$.}
\end{frame}

\begin{frame}{Якобиатор}
\begin{thmbox}{Лемма 9}
Пусть выполнены $(1),(2),(3),(5)$ (без тождества Якоби), и
$$J(a,b,c)=\{a,\{b,c\}\}-\{\{a,b\},c\}-(-1)^{\rp a\rp b}\{b,\{a,c\}\}.$$
Тогда
\begin{enumerate}
  \item[(i)] $J(a,b,c)=-(-1)^{\rp a\rp b}J(b,a,c)=-(-1)^{\rp b\rp c}J(a,c,b)$;
  \item[(ii)] $J(a,b,xy)=J(a,b,x)\,y+(-1)^{(|a|+|b|)|x|}\,x\,J(a,b,y)\;{\color{deepgreen}-\,J(a,b,1)\,xy}$.
\end{enumerate}
\end{thmbox}
\begin{itemize}\small
  \item Якобиатор — «дифференцирование с поправкой»; без поправки это классический факт для алгебр Пуассона.
  \item Доказательство: раскрыть три члена по $(5)$, лишние члены сокращаются попарно.
  \item Общая формула, может быть полезна сама по себе.
\end{itemize}
\end{frame}

\begin{frame}{Тождество Якоби и завершение}
\begin{enumerate}\setlength{\itemsep}{0.5em}
  \item \textbf{Предложение 10.} $J=0$ на $\Sm$:
  \begin{itemize}\small
    \item $J(u,v,w)=0$ при $u,v,w\in\Lh$ — это Якоби в $\LL$;
    \item по (ii) и индукции по числу множителей — для любого третьего аргумента;
    \item по (i) — для всех аргументов.
  \end{itemize}
  \item \textbf{Теорема 11.} $(\Sm,\cdot,\{\,,\}^*)$ — унитальная обобщённая алгебра Герстенхабера, порождённая $X$.
  \item \textbf{Теорема 12.} $\varphi\colon\G\to\Sm$ сюръективен, $\varphi(u)=u$ для $u\in U$, а $U$ — базис $\Sm$;
  вместе с леммой 5: $U$ — базис $\G$ и $\varphi$ — изоморфизм. \hfill$\square$
\end{enumerate}
\end{frame}

\partslide{05}{Размерность}

\begin{frame}{Мультилинейная часть}
\begin{itemize}\setlength{\itemsep}{0.45em}
  \item $X=\{1,x_1,\dots,x_n\}$; \ $\mathrm{GenGer}(n)$ — оболочка элементов $U$, где каждая из букв $1,x_1,\dots,x_n$ встречается ровно один раз.
  \item Это не однородная компонента: $1a=a$ и поправка в $(5)$ не однородны по $1$.
\end{itemize}
\vskip0.2cm
\begin{thmbox}{Теорема 14}
\centering\Large $\dim\mathrm{GenGer}(n)=n\cdot n!$
\end{thmbox}
\vskip0.2cm
\centering
\begin{tabular}{@{}lcccc@{}}
\toprule
$n$ & 1 & 2 & 3 & 4\\
\midrule
обобщённые алгебры Герстенхабера: $n\cdot n!$ & \textbf{1} & \textbf{4} & \textbf{18} & \textbf{96}\\
алгебры Герстенхабера ($\D=0$, по $x_1,\dots,x_n$): $n!$ & 1 & 2 & 6 & 24\\
\bottomrule
\end{tabular}
\vskip0.15cm
{\small\color{mute}Ответ $n\cdot n!$ совпадает с ответом для обобщённых супералгебр Пуассона (K17).}
\end{frame}

\begin{frame}{Доказательство: перестановки и циклы}
\begin{itemize}\setlength{\itemsep}{0.5em}
  \item Мультилинейный элемент $U$ $\leftrightarrow$ разбиение $\pi$ множества $X$ на блоки, \hl{без блока $\{1\}$}, и мультилинейное хорошее слово на каждом блоке.
  \item Хороших мультилинейных слов на $k$ буквах — $(k-1)!$ (размерность для свободной алгебры Ли).
  \item Квадраты $\{v,v\}$ не мультилинейны.
\end{itemize}
$$\dim\mathrm{GenGer}(n)=\sum_{\pi}\ \prod_{B\in\pi}(|B|-1)!$$
\begin{thmbox}{Подсчёт}
Разбиение $Y$ с циклическим порядком на каждом блоке $=$ перестановка $Y$ (блоки — циклы):
\quad $\sum_{\pi}\prod_B(|B|-1)!=|Y|!$.\\[0.3em]
Вычитаем разбиения с блоком $\{1\}$: \quad $(n+1)!-n!=n\cdot n!$.
\end{thmbox}
\end{frame}
