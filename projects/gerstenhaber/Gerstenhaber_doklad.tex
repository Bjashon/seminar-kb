\begin{frame}[label=poisson]{Алгебры Пуассона}
\small
\begin{defbox}{Алгебра Пуассона: пространство $V$ над $\F$, билинейные умножение $ab$ и скобка $\{a,b\}$}
\[ \renewcommand{\arraystretch}{1.3}\begin{array}{@{}l@{\hspace{1.4cm}}l@{}}
\begin{aligned}
ab&=ba\\
(ab)c&=a(bc)\\
1a&=a
\end{aligned}\quad\nm{умножение}
&
\begin{aligned}
\{a,b\}&=-\{b,a\}\\
\{a,\{b,c\}\}&=\{\{a,b\},c\}+\{b,\{a,c\}\}
\end{aligned}\quad\nm{скобка Ли}\\[16pt]
\multicolumn{2}{@{}l}{\{a,bc\}=\{a,b\}\,c+b\,\{a,c\}\qquad\nm{правило Лейбница}}
\end{array} \]
\end{defbox}
\begin{columns}[T,onlytextwidth]
\column{0.5\textwidth}
\only<2->{\begin{exbox}{Пример: $\F[p,q]$, \ $\{p,q\}=1$}
\[ \begin{aligned}
\{f,g\}&=\partial_pf\,\partial_qg-\partial_qf\,\partial_pg\\
\{p,q^2\}&=2q=\{p,q\}\,q+q\,\{p,q\}
\end{aligned} \]
\end{exbox}}
\column{0.46\textwidth}
\onslide<3->
\bul \hlt{Единицу скобка не видит:}
\[ \begin{aligned}
\{a,1\}&=\{a,1\cdot1\}=2\{a,1\}\\
\{a,1\}&=0
\end{aligned} \]
\onslide<4->
\bul Скобку достаточно знать на образующих: так на \ftip[-2.6cm]{sg}{$S(\mathfrak g)$}{S(g) — многочлены от базиса e₁, …, eₙ алгебры Ли g. Скобка Ли–Пуассона: {x, y} = [x, y] при x, y ∈ g, на произведениях — по правилу Лейбница.}{Многочлены от базиса алгебры Ли $\mathfrak g$; скобка задана на $\mathfrak g$ и продолжена по правилу Лейбница:
\[ \begin{aligned}
S(\mathfrak g)&=\F[e_1,\dots,e_n]\\
\{x,y\}&=[x,y],\quad x,y\in\mathfrak g\\
\{x,yz\}&=\{x,y\}\,z+y\,\{x,z\}
\end{aligned} \]} строится \hlt{скобка Ли--Пуассона}.
\end{columns}
\end{frame}

\begin{frame}[label=super]{Супералгебры}
\small
\begin{defbox}{Супералгебра — $\Z_2$-градуированная алгебра $V$ над $\F$, \ $\operatorname{char}\F=0$}
\[ V=V_0\oplus V_1\hspace{1.3cm}V_iV_j\subseteq V_{i+j}\hspace{1.3cm}|a|=i\ \ \text{при}\ a\in V_i\ \ \nm{чётность} \]
\end{defbox}
\onslide<2->
\[ ab=(-1)^{|a||b|}\,ba\qquad\nm{суперкоммутативность} \]
\only<3->{\begin{exbox}{Пример: суперполиномы $\F[x\mid\xi_1,\xi_2]$; \ $x$ чётная, \ $\xi_1,\xi_2$ нечётные}
\[ \begin{aligned}
V_0&=\spn\{x^k,\ x^k\xi_1\xi_2\} &\qquad \xi_2\xi_1&=-\xi_1\xi_2\\
V_1&=\spn\{x^k\xi_1,\ x^k\xi_2\} & \xi_1^2&=0
\end{aligned} \]
\end{exbox}}
\end{frame}

\begin{frame}[label=superder]{Супердифференцирования и супералгебры Ли}
\small
\begin{defbox}{Супердифференцирование чётности $p\in\Z_2$: линейное $d\colon V\to V$}
\[ \begin{aligned}
d(V_i)&\subseteq V_{i+p}\\
d(ab)&=d(a)\,b+(-1)^{p|a|}\,a\,d(b) &&\nm{знак: $d$ проходит мимо $a$}
\end{aligned} \]
\end{defbox}
\only<2->{\begin{exbox}{Пример: $d=\partial_{\xi_1}$ на $\F[x\mid\xi_1,\xi_2]$, \ $p=1$}
\[ \begin{aligned}
\partial_{\xi_1}(\xi_1\xi_2)&=\xi_2 &\qquad \partial_{\xi_1}(\xi_2\xi_1)&=(\partial_{\xi_1}\xi_2)\,\xi_1+(-1)^{1\cdot1}\,\xi_2\,\partial_{\xi_1}\xi_1=-\xi_2
\end{aligned} \]
\end{exbox}}
\only<3->{\begin{defbox}{Супералгебра Ли: скобка $[a,b]$ на $V$, \ $[V_i,V_j]\subseteq V_{i+j}$}
\[ \begin{aligned}
{}[a,b]&=-(-1)^{|a||b|}\,[b,a] &&\nm{суперкососимметричность}\\
{}[a,[b,c]]&=[[a,b],c]+(-1)^{|a||b|}\,[b,[a,c]] &&\nm{тождество Якоби}
\end{aligned} \]
\end{defbox}}
\end{frame}

\begin{frame}[label=gerst]{Алгебры Герстенхабера: нечётная скобка}
\small
\begin{defbox}{Алгебра Герстенхабера: суперкоммутативная ассоциативная супералгебра с $1$}
\[ \begin{aligned}
|\{a,b\}|&=|a|+|b|+1 &&\nm{скобка нечётна}\\
\rp a&=|a|+1 &&\nm{обращённая чётность}\\
\{a,b\}&=-(-1)^{\rp a\rp b}\{b,a\} &&\nm{суперкососимметричность}\\
\{a,\{b,c\}\}&=\{\{a,b\},c\}+(-1)^{\rp a\rp b}\{b,\{a,c\}\} &&\nm{тождество Якоби}\\
\{a,bc\}&=\{a,b\}c+(-1)^{\rp a|b|}b\{a,c\} &&\nm{правило Лейбница}
\end{aligned} \]
\end{defbox}
\only<2->{\begin{exbox}{Пример: $\Lambda(\mathfrak g)$, скобка Схоутена; \ $\mathfrak g=\langle h,e\rangle$, $[h,e]=2e$; элементы $\mathfrak g$ нечётны}
\[ \begin{aligned}
\{h,he\}&=\{h,h\}e+h\{h,e\}=2he\\
\{e,he\}&=\{e,h\}e+h\{e,e\}=-2e^2=0
\end{aligned} \]
\end{exbox}}
\onslide<3->
\textbf{Отличие от Пуассона:} для чётных $a,b$ скобка \hlt{симметрична}: бывает $\{a,a\}\neq0$.
\end{frame}

\begin{frame}[label=genpoisson]{Правило Лейбница с поправкой}
\small
В алгебре Пуассона скобка с единицей равна нулю. Что будет, если это разрешить?
\only<2->{\begin{defbox}{Обобщённая алгебра Пуассона (скобка чётная; Кантарини--Кац 2007)}
Скобка Ли и правило Лейбница с поправкой; $\{a,\cdot\}$ — дифференцирование плюс умножение на элемент:
\[ \{a,bc\}=\{a,b\}c+b\{a,c\}\;\mka{-\,\{a,1\}\,bc}. \]
\end{defbox}}
\only<3->{\begin{exbox}{Пример: $\F[x]$, скобка векторных полей $f\,\tfrac{d}{dx}$}
\[ \begin{aligned}
\{f,g\}&=fg'-f'g\\
\{f,1\}&=-f'\neq0\\
\{x,x^2\}&=x\cdot2x-1\cdot x^2=x^2\\
&=\{x,x\}\,x+x\,\{x,x\}-\{x,1\}\,x^2 &&\nm{$\{x,x\}=0$, \ $\{x,1\}=-1$}
\end{aligned} \]
\end{exbox}}
\onslide<4->
{\small\color{mute}Откуда поправка: йордановы скобки и \go{kantor}{двойник Кантора}.}
\end{frame}

\begin{frame}[label=freealg]{Свободная алгебра и её базис}
\small
\begin{itemize}\setlength{\itemsep}{0.3em}
  \item<1-> \hlt{Свободная алгебра} класса на буквах $X$: выражения от букв, отождествлённые только по тождествам класса. Любое отображение букв в алгебру класса продолжается до гомоморфизма.

  \item<3-> \hlt{Мультилинейная часть}: элементы, где каждая из букв $x_1,\dots,x_n$ входит ровно один раз. Её размерность — числовой «паспорт» класса.
\end{itemize}

\begin{frame}[label=good]{Алгебры Ли: хорошие слова Холла}
\small
\begin{defbox}{Хорошие слова: буквы и слова $w=[u,v]$, где $u,v$ — хорошие и}
\[ \begin{aligned}
u&>v &&\nm{порядок: по длине, затем лексикографически}\\
u_2&\le v\quad\text{при}\ u=[u_1,u_2]
\end{aligned} \]
\end{defbox}
\only<2->{\begin{exbox}{Пример: $x_1<x_2<x_3$, каждая буква по разу}
\[ \begin{aligned}
&[x_2,x_1]<[x_3,x_1]<[x_3,x_2] &&\nm{длина 2}\\
&[[x_2,x_1],x_3],\ \ [[x_3,x_1],x_2] &&\nm{хорошие длины 3}\\
&[[x_3,x_2],x_1]=[[x_3,x_1],x_2]-[[x_2,x_1],x_3] &&\nm{по Якоби}
\end{aligned} \]
\end{exbox}}
\only<3->{\begin{thmbox}{Теорема (М.~Холл, 1950)}
Хорошие слова — базис свободной алгебры Ли; мультилинейных на $k$ буквах — $(k-1)!$.
\end{thmbox}}
\end{frame}

\begin{frame}[label=superlie]{Супералгебры Ли: квадраты нечётных}
\small
Для нечётного $v$ кососимметричность ничего не требует — квадрат \hlt{не обязан} быть нулём:
\[ [v,v]=-(-1)^{1\cdot1}[v,v]=[v,v]. \]
\only<2->{\begin{exbox}{Пример: $\mathfrak{gl}(1|1)$, \ $v=E_{12}+E_{21}$ нечётен; для нечётных $[a,b]=ab+ba$}
\[ [v,v]=2v^2=2(E_{12}E_{21}+E_{21}E_{12})=2(E_{11}+E_{22})\neq0 . \]
\end{exbox}}
\onslide<3->
По Якоби из квадратов ничего нового не растёт; квадраты нужны только как отдельные элементы базиса:
\[ \begin{aligned}
{}[v,[v,v]]&=0\\
{}[[v,v],w]&=2\,[v,[v,w]]
\end{aligned} \]
\only<4->{\begin{thmbox}{Теорема (А.\,С.~Штерн, 1986)}
Базис свободной супералгебры Ли: хорошие слова и \hlt{квадраты $[v,v]$ нечётных хороших слов $v$}.
\end{thmbox}}
\end{frame}

\begin{frame}[label=shestakov]{Алгебры Пуассона: шаблон Шестакова}
\small
\begin{thmbox}{Теорема (И.\,П.~Шестаков, 1993)}
Свободная алгебра Пуассона на $X$ — это $S(\mathrm{Lie}(X))$: базис — \hlt{одночлены от хороших слов}.
\end{thmbox}
\only<2->{\begin{exbox}{Пример: мультилинейные элементы от $x_1<x_2<x_3$; \ всего $1+3+2=6=3!$}
\[ \underbrace{x_1x_2x_3}_{1},\quad
\underbrace{\{x_2,x_1\}x_3,\ \{x_3,x_1\}x_2,\ \{x_3,x_2\}x_1}_{3},\quad
\underbrace{\{\{x_2,x_1\},x_3\},\ \{\{x_3,x_1\},x_2\}}_{2} \]
\end{exbox}}
\onslide<3->
\hlt{Порождение}: Лейбниц выносит произведения из скобок. \hlt{Независимость}: модель $S(\mathrm{Lie}(X))$.
\[ \{x_1,x_2^2\}=\{x_1,x_2\}x_2+x_2\{x_1,x_2\}=-2x_2\{x_2,x_1\} \]
\onslide<4->
\hlt{Почему $n!$}: блоки букв и хорошее слово на каждом; слов на $k$ буквах $(k-1)!$, как циклов.
\end{frame}

\begin{sidecols}

\end{sidebox}
\maincol
\begin{exbox}{Пример: $\{x_1,x_2x_3\}$ по (5); буквы чётные, $\rp{x_i}=1$}
\[ \begin{aligned}
\{x_1,x_2x_3\}&=\{x_1,x_2\}x_3+x_2\{x_1,x_3\}-\{1,x_1\}x_2x_3\\

Слева буквы $1$ нет, а в ответе появилось слово $\{x_1,1\}$.

Значит, $1$ — \hlt{буква скобки}:

\end{aligned} \]

\end{sidecols}

Для независимости нужна алгебра нашего класса, где $U$ — базис по построению. Берём суперполиномы $\Sm$ из главной теоремы: в них одночлены $U$ независимы.\par\medskip
Остаётся задать на $\Sm$ скобку. Она будет задана явной формулой: каждый множитель с каждым и поправки, которые требует правило (5).}
\only<3| handout:3>{{\large\bfseries Проверяем аксиомы\par}\vskip0.2cm\raggedright
Проверяем, что $\Sm$ с новой скобкой удовлетворяет аксиомам (1)--(5). Кососимметричность (3) и правило (5) проверяются прямо: формула скобки замкнутая.\par\medskip
Самое трудное — тождество Якоби (4). Для него понадобится мера его нарушения, якобиатор.}
\only<4| handout:4>{{\large\bfseries $U$ — базис $\G$\par}\vskip0.2cm\raggedright
По свойству свободной алгебры есть гомоморфизм $\G\to\Sm$, тождественный на $X$, а значит, и на $U$. Комбинация, равная нулю в $\G$, равна нулю и в $\Sm$:
\[ \sum_u c_u\,u=0\ \text{в}\ \G\ \Longrightarrow\ \sum_u c_u\,u=0\ \text{в}\ \Sm\ \Longrightarrow\ c_u=0 . \]
Значит, $U$ линейно независимо, и вместе с первым этапом $U$ — базис $\G$.}
\end{columns}

\begin{frame}[label=kantor]{Откуда поправка: йордановы скобки и двойник Кантора}
\begin{defbox}{Двойник Кантора $K(A)=A\oplus Ax$}
$A$ — суперкоммутативная супералгебра со скобкой $\{\,,\}$, $Ax$ — копия $A$; умножение:
$$a\ast b=ab,\qquad a\ast bx=(ab)x,\qquad ax\ast b=(-1)^{|b|}(ab)x,\qquad ax\ast bx=(-1)^{|b|}\{a,b\}.$$
\end{defbox}
\begin{itemize}\small\setlength{\itemsep}{0.35em}
  \item Кантор (1990): для скобки Пуассона $K(A)$ — \ftip[-4.4cm]{jordan}{йорданова супералгебра}{суперкоммутативная алгебра с тождеством Йордана (x²y)x = x²(yx), с суперзнаками}{Суперкоммутативная супералгебра с тождеством Йордана; для чётных $x,y$
\[ (x^2y)\,x=x^2\,(yx), \]
для остальных — со знаками по правилу знаков.}.
  \item Кинг--МакКриммон (1992): $K(A)$ йорданова $\iff$ $\{\,,\}$ — \hlt{йорданова скобка}: Лейбниц и Якоби выполнены с поправками через $\{a,1\}$.
  \item Замена $\{a,b\}\mapsto\{a,b\}+\tfrac12\bigl(a\{b,1\}-\{a,1\}b\bigr)$ делает йорданову скобку обобщённой пуассоновой: Якоби точное, поправка только в Лейбнице.
  \item Нечётный вариант — \hlt{обобщённые алгебры Герстенхабера} (Кантарини--Кац 2010).
\end{itemize}
\end{frame}

\begin{frame}[label=po]{\ft{Примеры с $\D\neq0$: $PO(n,n+1)$}{Примеры с ненулевым D: PO(n,n+1)}}
\small
К $PO(n,n)$ добавим нечётную переменную $\tau$: $PO(n,n+1)=\F[x_1,\dots,x_n\mid\xi_1,\dots,\xi_n,\tau]$ со скобкой ($\{\,,\}_H$ — скобка Баттина, $E$ — оператор Эйлера)
$$\{f,g\}_K=\{f,g\}_H+(E-2)(f)\,\frac{\partial g}{\partial\tau}+(-1)^{|f|}\frac{\partial f}{\partial\tau}\,(E-2)(g),\qquad

\begin{frame}[label=jacobi]{Якобиатор}
\begin{thmbox}{Лемма 9}
Пусть выполнены \ax{1}, \ax{2}, \ax{3}, \ax{5} (без тождества Якоби), и
$$J(a,b,c)=\{a,\{b,c\}\}-\{\{a,b\},c\}-(-1)^{\rp a\rp b}\{b,\{a,c\}\}.$$
Тогда
\begin{enumerate}
  \item[(i)] $J(a,b,c)=-(-1)^{\rp a\rp b}J(b,a,c)=-(-1)^{\rp b\rp c}J(a,c,b)$;
  \item[(ii)] $J(a,b,xy)=J(a,b,x)\,y+(-1)^{(|a|+|b|)|x|}\,x\,J(a,b,y)\;{\color{deepgreen}-\,J(a,b,1)\,xy}$.
\end{enumerate}
\end{thmbox}
\begin{itemize}\small

  \item Общая формула, может быть полезна сама по себе.
\end{itemize}
\end{frame}

\begin{frame}[label=formulas1]{Формулы доклада: алгебры Пуассона и супералгебры}
\small
\fe{Алгебра Пуассона}{\go{poisson}{слайд «Алгебры Пуассона»}}{Скобка с первым аргументом $a$ — дифференцирование, поэтому скобка с единицей равна нулю.}
\[ \begin{aligned}
\{a,bc\}&=\{a,b\}\,c+b\,\{a,c\} &&\nm{правило Лейбница}\\
\{a,1\}&=0
\end{aligned} \]
\fe{Супералгебра}{\go{super}{слайд «Супералгебры»}}{$\Z_2$-градуированная алгебра; $|a|$ — чётность.}
\[ \begin{aligned}
V&=V_0\oplus V_1\hspace{1.3cm}V_iV_j\subseteq V_{i+j}\\
ab&=(-1)^{|a||b|}\,ba &&\nm{суперкоммутативность}
\end{aligned} \]
\fe{Супердифференцирование}{\go{superder}{слайд «Супердифференцирования»}}{Линейное $d\colon V\to V$ чётности $p$.}
\[ \begin{aligned}
d(V_i)&\subseteq V_{i+p}\\
d(ab)&=d(a)\,b+(-1)^{p|a|}\,a\,d(b)
\end{aligned} \]
\end{frame}

\begin{frame}[label=formulas1b]{Формулы доклада: алгебры Герстенхабера}
\small
\fe{Супералгебра Ли}{\go{superder}{слайд «Супералгебры Ли»}}{Скобка складывает чётности.}
\[ \begin{aligned}
{}[V_i,V_j]&\subseteq V_{i+j}\\
{}[a,b]&=-(-1)^{|a||b|}\,[b,a]\\
{}[a,[b,c]]&=[[a,b],c]+(-1)^{|a||b|}\,[b,[a,c]]
\end{aligned} \]
\fe{Алгебра Герстенхабера}{\go{gerst}{слайд «Алгебры Герстенхабера»}}{Скобка нечётна.}
\[ \begin{aligned}
|\{a,b\}|&=|a|+|b|+1\\
\rp a&=|a|+1\\
\{a,b\}&=-(-1)^{\rp a\rp b}\{b,a\}\\
\{a,\{b,c\}\}&=\{\{a,b\},c\}+(-1)^{\rp a\rp b}\{b,\{a,c\}\}\\
\{a,bc\}&=\{a,b\}c+(-1)^{\rp a|b|}b\{a,c\}
\end{aligned} \]
\fe{Поправка в правиле Лейбница}{\go{genpoisson}{слайд «Правило Лейбница с поправкой»}}{Скобка чётная.}
\[ \{a,bc\}=\{a,b\}c+b\{a,c\}\;\mka{-\,\{a,1\}\,bc} \]
\end{frame}

\begin{frame}[label=formulas2]{Формулы доклада: определение 1}
\small
\fe{Обобщённая алгебра Герстенхабера}{\go{def1}{слайд «Определение 1»}}{$\Z_2$-градуированное пространство с $1\in V_0$, умножением и скобкой; $\D(a)=\{1,a\}$.}
\hypertarget{f:5}{}
\[ \begin{aligned}
&\ano{1} & |ab|&=|a|+|b|\\
&& |\{a,b\}|&=|a|+|b|+1\\
&\ano{2} & (ab)c&=a(bc)\\
&& ab&=(-1)^{|a||b|}ba\\
&& 1a&=a\\
&\ano{3} & \{a,b\}&=-(-1)^{\rp a\rp b}\{b,a\}\\
&\ano{4} & \{a,\{b,c\}\}&=\{\{a,b\},c\}+(-1)^{\rp a\rp b}\{b,\{a,c\}\}\\
&\ano{5} & \{a,bc\}&=\{a,b\}c+(-1)^{\rp a|b|}b\{a,c\}\;\mka{+(-1)^{\rp a}\D(a)\,bc}
\end{aligned} \]
\fe{Единица в (5)}{\go{ex1}{слайд «Новое явление»}}{$\D$ — дифференцирование ровно тогда, когда $\{1,1\}=0$.}
\[ \D(bc)=\D(b)\,c+(-1)^{|b|}b\,\D(c)\;\mka{-\,\D(1)\,bc} \]
\end{frame}

\begin{frame}[label=formulas3a]{Формулы доклада: задача}
\small
\fe{Хорошие слова}{\go{good}{слайд «Хорошие слова Холла»}}{Порядок — по длине, затем лексикографически; слово $w=[u,v]$ хорошее, если}
\[ \begin{aligned}
u,v&\ \text{— хорошие}\\
u&>v\\
u_2&\le v\quad\text{при}\ u=[u_1,u_2]
\end{aligned} \]
\fe{Квадраты нечётных}{\go{superlie}{слайд «Супералгебры Ли»}}{Других новых элементов квадраты не порождают.}
\[ \begin{aligned}
{}[v,[v,v]]&=0\\
{}[[v,v],w]&=2\,[v,[v,w]]
\end{aligned} \]
\fe{Единица становится буквой}{\go{letter1}{слайд «Единица — буква»}}{Буквы чётные; в последнем члене появилась буква $1$.}
\[ \{x_1,x_2x_3\}=x_3\{x_2,x_1\}+x_2\{x_3,x_1\}\;\mka{-\,x_2x_3\{x_1,1\}} \]
\end{frame}

\begin{frame}[label=formulas3]{Формулы доклада: ответ}
\small
\fe{Скобочная часть}{\go{setM}{слайд «Множество $M$»}}{Квадраты берутся у чётных слов.}
\[ \begin{aligned}
M&=\{\,u,\ \{v,v\}\mid u,v\ \text{— хорошие слова над}\ X,\ |v|=0\,\}\\
e_1&=1<e_2<e_3<\cdots &&\nm{элементы $M$ по порядку}
\end{aligned} \]
\fe{Базис и модель}{\go{thm12}{слайд «Главная теорема»}}{Одночлены без множителя $1$, нечётные — в первой степени; модель — суперполиномы.}
\[ \begin{aligned}
U&=\{\,e_{i_1}^{k_1}\cdots e_{i_r}^{k_r}\mid 1<i_1<\dots<i_r,\ \ k_j=1\ \text{при}\ |e_{i_j}|=1\,\}\\
\Sm&=S(\LL')\\
\LL'&=\spn(M\setminus\{1\})
\end{aligned} \]
\end{frame}

\begin{frame}[label=formulas4]{Формулы доклада: доказательство}

\fe{Модель}{\go{formula8}{слайд «Формулы (7) и (8)»}}{Множители $a_i$, $b_j$ берутся из $\Lh=\F1\oplus\LL'$; формулы задают $\D$ и скобку на произведениях.}

\D(a_1\cdots a_m)&=\sum_i(-1)^{|a_{<i}|}a_{<i}\{1,a_i\}a_{>i}-(m-1)\{1,1\}\,a \tag{7}\\
\{a,b\}^*&=\mka{\sum_{i,j}(-1)^{|a_i||a_{>i}|+|b_j||b_{<j}|}a_{\hat\imath}\{a_i,b_j\}b_{\hat\jmath}}\\

\end{align*}
\end{frame}
