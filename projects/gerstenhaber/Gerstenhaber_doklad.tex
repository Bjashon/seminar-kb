\begin{frame}[label=poisson]{Алгебры Пуассона}
\small
\begin{defboxn}
\hlt{Алгебра Пуассона} — в.п. $V$ над $\F$ с единицей $1$, билинейными умножением $ab$ и скобкой $\{a,b\}$, такими что для всех $a,b,c$
\[ \renewcommand{\arraystretch}{1.3}\begin{array}{@{}l@{\hspace{0.7cm}}l@{}}
\ano{1.1}\begin{aligned}
ab&=ba\\
(ab)c&=a(bc)\\
1a&=a
\end{aligned}\quad\nm{умножение}
&
\ano{1.2}\begin{aligned}
\{a,b\}&=-\{b,a\}\\
\{a,\{b,c\}\}&=\{\{a,b\},c\}+\{b,\{a,c\}\}
\end{aligned}\quad\nm{скобка Ли}\\[4pt]
\multicolumn{2}{c}{\phantom{\nm{правило Лейбница}\qquad}\ano{1.3}\{a,bc\}=\{a,b\}\,c+b\,\{a,c\}\qquad\nm{правило Лейбница}}
\end{array} \]
\end{defboxn}
\begin{columns}[T,onlytextwidth]
\column{0.5\textwidth}
\only<2->{\begin{exbox}{Пример: $\F[p,q]$, \ \ftip[-6.3cm]{pq}{$\{f,g\}=\partial_pf\,\partial_qg-\partial_qf\,\partial_pg$}{Это скобка Пуассона: кососимметрична (f и g меняются местами); правило Лейбница — потому что ∂p и ∂q дифференцирования, ∂(gh) = ∂g·h + g·∂h; тождество Якоби — прямой счёт.}{Почему это скобка Пуассона:
\[ \begin{aligned}
\{g,f\}&=-\{f,g\}\\
\partial(gh)&=\partial g\,h+g\,\partial h &&\nm{$\partial=\partial_p,\ \partial_q$}\\
\{f,gh\}&=\{f,g\}\,h+g\,\{f,h\}
\end{aligned} \]
Тождество Якоби — прямой счёт: члены со вторыми производными сокращаются.}}
\[ \begin{aligned}
\{p,q\}&=\partial_pp\,\partial_qq-\partial_qp\,\partial_pq=1\cdot1-0\cdot0=1\\
\{p,q^2\}&=\{p,q\}\,q+q\,\{p,q\}=2q
\end{aligned} \]
\end{exbox}}
\column{0.46\textwidth}
\onslide<3->
\bul \hlt{Скобка с единицей равна нулю:}
\[ \begin{aligned}
\{a,1\}&=\{a,1\cdot1\}=2\{a,1\}\\
\{a,1\}&=0
\end{aligned} \]
\onslide<4->
\bul Скобка однозначно задаётся значениями на образующих; так на \ftip[-2.6cm]{sg}{$S(\mathfrak g)$}{S(g) — многочлены от базиса e₁, …, eₙ алгебры Ли g. Скобка Ли–Пуассона на базисе — скобка g: {eᵢ, eⱼ} = Σ cᵢⱼᵏ eₖ, где cᵢⱼᵏ — структурные константы g; на произведениях — по правилу Лейбница. \[ \begin{aligned}
S(\mathfrak g)&=\F[e_1,\dots,e_n]\\
\{e_i,e_j\}&=\textstyle\sum_k c_{ij}^k\,e_k\\
\{x,yz\}&=\{x,y\}\,z+y\,\{x,z\}\\
\{f,g\}&=\sum_{i,j,k}c_{ij}^k\,e_k\,\frac{\partial f}{\partial e_i}\,\frac{\partial g}{\partial e_j} &&\nm{явная формула}
\end{aligned} \]} задаётся \hlt{скобка Ли--Пуассона}.
\end{columns}
\end{frame}

\begin{frame}[label=super]{Супералгебры}
\small
\begin{defboxn}
Пусть $\F$ — поле, $\operatorname{char}\F=0$. \hlt{Супералгебра} — в.п. $V$ над $\F$ с билинейным умножением $ab$ и \hlt{$\Z_2$-градуировкой} — разложением в прямую сумму, согласованным с умножением:
\[ V=V_0\oplus V_1,\qquad V_iV_j\subseteq V_{(i+j)\bmod 2}\quad (i,j\in\{0,1\}). \]
Элементы $V_0$ — \hlt{чётные}, $V_1$ — \hlt{нечётные}, те и другие — \hlt{однородные}; \hlt{чётность} $|a|=i$ при $a\in V_i$. Всякий $a$ однозначно раскладывается: $a=a_0+a_1$, $a_i\in V_i$.
\end{defboxn}
\vspace{-2pt}\noindent{\color{applegreen}\large$\Rightarrow$}\enspace Далее $V=V_0\oplus V_1$ — супералгебра, $a,b,c$ в формулах с чётностями — однородные.\par\vspace{4pt}
\only<2->{\begin{exbox}{Пример: матрицы $2\times2$; \ $V_0$ — диагональные, \ $V_1$ — внедиагональные}
\[ \begin{aligned}
E_{11}E_{12}&=E_{12}\in V_1 &&\nm{$E_{ij}$ — матричные единицы; \ $0+1=1$}\\
E_{12}E_{21}&=E_{11}\in V_0 &&\nm{$1+1=0$ по модулю 2}
\end{aligned} \]
\end{exbox}}
\end{frame}

\begin{frame}[label=supercomm]{Суперкоммутативные супералгебры}
\small
\begin{defboxn}
$V$ \hlt{суперкоммутативна}, если для всех $a,b$
\[ \ano{1.4}ab=(-1)^{|a||b|}\,ba\qquad\nm{правило Кошуля: $a\,b\to(-1)^{|a||b|}\,b\,a$} \]
Значит, чётный элемент коммутирует со всеми, а нечётные антикоммутируют; в частности, при $|a|=1$ по (1.4) $a^2=-a^2$, то есть $a^2=0$.
\end{defboxn}
\only<2->{\begin{exbox}{Пример: матрицы $2\times2$ — супералгебра, но не суперкоммутативная}
\[ \begin{aligned}
E_{12}E_{21}&=E_{11} &&\nm{$E_{12}$, $E_{21}$ нечётные}\\
-E_{21}E_{12}&=-E_{22}\neq E_{11} &&\nm{(1.4) не выполнено}
\end{aligned} \]
\end{exbox}}
\end{frame}

\begin{frame}[label=superder]{Супердифференцирования}
\small
\begin{defboxn}
\hlt{Супердифференцирование чётности $p\in\{0,1\}$} — линейное $d\colon V\to V$, сдвигающее чётность на $p$,
\[ d(V_i)\subseteq V_{(i+p)\bmod 2},\qquad\text{то есть}\qquad |d(a)|=(|a|+p)\bmod 2, \]
и удовлетворяющее правилу Лейбница со знаком:
\[ \ano{1.5}d(ab)=d(a)\,b+(-1)^{p|a|}\,a\,d(b)\qquad\nm{правило Кошуля: $d\,a\to(-1)^{p|a|}\,a\,d$} \]
При $p=0$ это обычное дифференцирование.
\end{defboxn}
\only<2->{\begin{exbox}{Пример: на $\F[x\mid\xi_1,\xi_2]$ производная $d=\partial_{\xi_1}$, \ $p=1$}
\[ \begin{aligned}
\partial_{\xi_1}(\xi_1\xi_2)&=\xi_2\\
\partial_{\xi_1}(\xi_2\xi_1)&=(\partial_{\xi_1}\xi_2)\,\xi_1+(-1)^{1\cdot1}\,\xi_2\,\partial_{\xi_1}\xi_1=-\xi_2 &&\nm{по \fn{1.5}}
\end{aligned} \]
\end{exbox}}
\end{frame}

\begin{frame}[label=freealg]{Свободная алгебра и её базис}
\small
\begin{defboxn}
Пусть $\mathcal K$ — класс алгебр, заданный тождествами (например, алгебры Ли), $X$ — множество (для супералгебр — с чётностями элементов).\par\smallskip
\hlt{Свободная алгебра} класса $\mathcal K$ на $X$ — алгебра $F(X)\in\mathcal K$, порождённая $X$, такая что для любой $A\in\mathcal K$ и любого отображения $\varphi$, сохраняющего чётность,
\[ \varphi\colon X\to A\quad\Longrightarrow\quad\exists!\ \text{гомоморфизм}\ \Phi\colon F(X)\to A,\qquad \Phi|_X=\varphi. \]
Элементы $F(X)$ — выражения от букв, равные только в силу тождеств класса.
\end{defboxn}
\only<2->{\begin{exbox}{Пример: $\mathcal K$ — ассоциативные коммутативные алгебры с $1$, \ $X=\{x,y\}$; \ $F(X)=\F[x,y]$}
\[ \begin{aligned}
\varphi\colon\ &x\mapsto t^2,\ \ y\mapsto t+1 &&\nm{любое отображение в $A=\F[t]$}\\
\Phi(f)&=f(t^2,\ t+1) &&\nm{гомоморфизм — подстановка}\\
\Phi(x^2y+3)&=t^4(t+1)+3 &&\nm{единствен: $\Phi(x^ay^b)=\varphi(x)^a\varphi(y)^b$}
\end{aligned} \]
\end{exbox}}
\end{frame}

\begin{frame}[label=freealgtab]{Базис и мультилинейная часть}
\small
\begin{defboxn}
\hlt{Базис} свободной алгебры — явно описанный базис $F(X)$ как в.п., то есть нормальная форма: равенство элементов проверяется сравнением коэффициентов. \hlt{Мультилинейная часть} — линейная оболочка элементов $F(x_1,\dots,x_n)$, в которые каждая буква входит ровно один раз.
\end{defboxn}
\onslide<2->
\vskip0.1cm
{\small\renewcommand{\arraystretch}{1.3}
\begin{tabular}{@{}L{3.4cm}L{7.6cm}cc@{}}
\toprule

\begin{frame}[label=good]{Алгебры Ли: хорошие слова Холла}
\small
\begin{defboxn}
Пусть $X$ — упорядоченное множество букв; \hlt{слова} — буквы и $[u,v]$ от слов $u,v$, \hlt{степень} — число букв. \hlt{Хорошие слова} и порядок на них — индукцией по степени:
\begin{itemize}\setlength{\itemsep}{0pt}
  \item буквы $x\in X$; порядок — как в $X$;
  \item слова $w=[u,v]$, где $u,v$ — хорошие и
  \[ \ano{2.1}u>v\quad\text{и}\quad\ano{2.2}[[u_1,u_2],v]:\ \ u_2\le v. \]
\end{itemize}
Порядок: большая степень — больше; при равной $[u,v]>[u',v']\iff u>u'$ или $u=u'$, $v>v'$.
\end{defboxn}
\only<2->{\begin{exbox}{Пример: $x_1<x_2<x_3$, каждая буква по разу}
\[ \begin{aligned}
&[x_2,x_1]<[x_3,x_1]<[x_3,x_2];\qquad [[x_2,x_1],x_3],\ \ [[x_3,x_1],x_2] &&\nm{хорошие слова степеней 2 и 3}\\
&[[x_3,x_2],x_1]=[[x_3,x_1],x_2]-[[x_2,x_1],x_3] &&\nm{не хорошее: $x_2>x_1$; по Якоби}
\end{aligned} \]
\end{exbox}}
\onslide<3->
\textbf{Теорема М.~Холла}\lit{M.~Hall, \emph{A basis for free Lie rings and higher commutators in free groups}, Proc. AMS \textbf{1} (1950), 575--581.}\textbf{.} Хорошие слова — базис свободной алгебры Ли; мультилинейных на $k$ буквах $(k-1)!$.
\end{frame}

\begin{frame}[label=superlie]{Супералгебры Ли: квадраты нечётных}
\small
Для нечётного $v$ суперкососимметричность при $a=b=v$ выполняется тождественно, и квадрат $[v,v]$ \hlt{может быть ненулевым}:
\[ [v,v]=-(-1)^{1\cdot1}[v,v]=[v,v]. \]
\only<2->{\begin{exbox}{Пример: $\mathfrak{gl}(1|1)$, \ $v=E_{12}+E_{21}$ нечётен; для нечётных $[a,b]=ab+ba$}
\[ [v,v]=2v^2=2(E_{12}E_{21}+E_{21}E_{12})=2(E_{11}+E_{22})\neq0 . \]
\end{exbox}}
\onslide<3->
По тождеству Якоби скобки, содержащие квадрат, выражаются через скобки без него; поэтому квадраты входят в базис только сами по себе:
\[ [v,[v,v]]=0,\qquad\qquad [[v,v],w]=2\,[v,[v,w]]. \]
\only<4->{\begin{thmbox}{Теорема (А.\,С.~Штерн, 1986)}
Базис свободной супералгебры Ли — хорошие слова и \hlt{квадраты $[v,v]$ нечётных хороших $v$}.\lit{A.~Shtern, \emph{Free Lie superalgebras}, Sib. Math. J. \textbf{27} (1986), 136--140.}
\end{thmbox}}
\end{frame}

\begin{frame}[label=setM]{\ft{Скобочная часть: множество $M$}{Скобочная часть: множество M}}
\small
Буквы — $X=\{1<x_1<x_2<\cdots\}$ с заданными чётностями, $|1|=0$.

Слова над $X$ — буквы и $\{u,v\}$ от слов $u,v$; хорошие слова и порядок — \go{good}{как у Холла}.\\
Скобка нечётная:
\[ |\{u,v\}|=|u|+|v|+1. \]
\begin{defboxn}
\hlt{Множество $M$} состоит из хороших слов и квадратов $\{v,v\}$ хороших слов чётности $0$:
\[ M=\bigl\{\,u,\ \{v,v\}\ \bigm|\ u,v\ \text{— хорошие слова над}\ X,\ |v|=0\,\bigr\}. \]
Квадраты сравниваются с остальными элементами так же, как слова. По возрастанию:
\[ M=\{e_1=1<e_2<e_3<\cdots\}. \]
\end{defboxn}
\only<2->{\begin{thmbox}{Лемма 4}
$M$ — базис свободной \ftip[-5.2cm]{oddlie}{нечётной супералгебры Ли}{кососимметричность (3) и Якоби (4): супералгебра Ли относительно чётности ‖a‖ = |a| + 1}{Супералгебра Ли относительно обращённой чётности $\rp a=|a|+1$:
\[ \begin{aligned}
\{a,b\}&=-(-1)^{\rp a\rp b}\{b,a\} &&\nm{3}\\
\{a,\{b,c\}\}&=\{\{a,b\},c\}+(-1)^{\rp a\rp b}\{b,\{a,c\}\} &&\nm{4}
\end{aligned} \]} $\LL$ на $X$.
\end{thmbox}}
\onslide<3->
\textbf{Доказательство.} $\LL$ — супералгебра Ли для чётности $\rp a=|a|+1$. По \go{superlie}{теореме Штерна} её базис — хорошие слова и квадраты $\{v,v\}$ хороших слов с $\rp v=1$, т.\,е. с $|v|=0$. Это в точности $M$.\par
\textbf{В частности,} $\{1,1\}\in M$, так как $|1|=0$; значит, $\{1,1\}\neq0$ в $\LL$.
\end{frame}

\begin{frame}[label=multi]{\ft{Мультилинейная часть: $n\cdot n!$}{Мультилинейная часть: n·n!}}
\small
\begin{defboxn}
Пусть $X=\{1,x_1,\dots,x_n\}$. \hlt{Мультилинейная часть} $\mathrm{GenGer}(n)$ — линейная оболочка тех элементов $U$, в которые каждая из букв $1,x_1,\dots,x_n$ входит ровно один раз; \hlt{буква $1$ тоже считается}.
\end{defboxn}
\only<2->{\begin{thmbox}{Теорема 14}
\centering\Large $\dim\mathrm{GenGer}(n)=n\cdot n!$
\end{thmbox}}
\onslide<3->
\vskip0.1cm
\centering\small\renewcommand{\arraystretch}{1.2}
\begin{tabular}{@{}lcccc@{}}
\toprule
$n$ & 1 & 2 & 3 & 4\\
\midrule
обобщённые алгебры Герстенхабера (и Пуассона, K17): $n\cdot n!$ & \textbf{1} & \textbf{4} & \textbf{18} & \textbf{96}\\
алгебры Пуассона и Герстенхабера, по $x_1,\dots,x_n$: $n!$ & 1 & 2 & 6 & 24\\
\bottomrule
\end{tabular}
\vskip0.25cm
\raggedright
{\small Сначала — тот же подсчёт для алгебр Пуассона (\go{shestakovn}{$n!$}), затем \go{perm}{доказательство} с примером $n=2$.}
\end{frame}

\begin{frame}[label=shestakovn]{\ft{Алгебры Пуассона: почему $n!$}{Алгебры Пуассона: почему n!}}
\small
$P_n$ — мультилинейная часть свободной алгебры Пуассона: $x_1,\dots,x_n$ входят по одному разу.

Базис $P_n$ — одночлены $u_1\cdots u_r$ от хороших слов (\go{shestakov}{Шестаков}). $B_j$ — множество букв слова $u_j$; $\{B_1,\dots,B_r\}$ — разбиение множества $\{x_1,\dots,x_n\}$.
\[ \begin{aligned}
\dim P_n&=\sum_{\{B_1,\dots,B_r\}}\ \prod_{j=1}^{r}\#\{\text{хорошие слова на буквах}\ B_j\} &&\nm{\go{shestakov}{базис Шестакова}}\\
&=\sum_{\{B_1,\dots,B_r\}}\ \prod_{j=1}^{r}\bigl(|B_j|-1\bigr)! &&\nm{\go{good}{теорема Холла}}\\
&=|S_n|=n! &&\nm{$B_j$ — циклы $\sigma$; циклов на $B$ — $(|B|-1)!$}
\end{aligned} \]
\only<2->{\begin{exbox}{Пример: $n=3$ — шесть элементов из \go{shestakov}{примера к теореме Шестакова}}
\[ \underbrace{(3-1)!}_{\{x_1,x_2,x_3\}}+\underbrace{3\cdot1!\cdot0!}_{\{x_i,x_j\}\sqcup\{x_k\}}+\underbrace{0!\cdot0!\cdot0!}_{\{x_1\}\sqcup\{x_2\}\sqcup\{x_3\}}=2+3+1=6=3! \]
\end{exbox}}
\end{frame}

\begin{frame}[label=perm]{Доказательство теоремы 14}
\small
Элемент $U$, где каждая из букв $1,x_1,\dots,x_n$ входит один раз, — одночлен $u_1\cdots u_r$ от хороших слов: в квадрат $\{v,v\}$ буквы $v$ входят дважды.

$B_j$ — множество букв слова $u_j$; $\{B_1,\dots,B_r\}$ — разбиение множества $\{1,x_1,\dots,x_n\}$, и ${\color{deepgreen}B_j\neq\{1\}}$: множителя $1$ в $U$ нет.
\[ \begin{aligned}
\dim\mathrm{GenGer}(n)&=\sum_{\{B_1,\dots,B_r\},\ B_j\neq\{1\}}\ \prod_{j=1}^{r}\bigl(|B_j|-1\bigr)! &&\nm{\go{good}{теорема Холла}}\\
&=\#\{\sigma\in S_{n+1}\mid\sigma(1)\neq1\} &&\nm{$B_j$ — циклы $\sigma$, \go{shestakovn}{как для $n!$}}\\
&=(n+1)!-n!=n\cdot n!
\end{aligned} \]
\only<2->{\begin{exbox}{Пример: $n=2$, буквы $1,x_1,x_2$, \ $3!=6$ перестановок}
\small\renewcommand{\arraystretch}{1.15}
\begin{tabular}{@{}l@{\qquad}l@{}}
$(1\,x_1)(x_2)$, \ $(1\,x_2)(x_1)$ & $\{x_1,1\}\,x_2$, \ $\{x_2,1\}\,x_1$\\
$(1\,x_1\,x_2)$, \ $(1\,x_2\,x_1)$ & $\{\{x_1,1\},x_2\}$, \ $\{\{x_2,1\},x_1\}$\\
{\color{mute}$(1)(x_1)(x_2)$, \ $(1)(x_1\,x_2)$} & {\color{mute}$1$ неподвижна — не считаются}\\
\end{tabular}\\[0.3em]
Итого $6-2=4=2\cdot2!$.
\end{exbox}}
\end{frame}

\begin{frame}[label=kantor]{Откуда поправка: йордановы скобки и двойник Кантора}
\begin{defbox}{Двойник Кантора $K(A)=A\oplus Ax$}
$A$ — суперкоммутативная супералгебра со скобкой $\{\,,\}$, $Ax$ — копия $A$; умножение:
$$a\ast b=ab,\qquad a\ast bx=(ab)x,\qquad ax\ast b=(-1)^{|b|}(ab)x,\qquad ax\ast bx=(-1)^{|b|}\{a,b\}.$$
\end{defbox}
\begin{itemize}\small\setlength{\itemsep}{0.35em}
  \item Кантор (1990)\lit{I.~Kantor, \emph{Jordan and Lie superalgebras determined by a Poisson algebra}, Amer. Math. Soc. Transl. \textbf{151} (1992), 55--80.}: для скобки Пуассона $K(A)$ — \ftip[-4.4cm]{jordan}{йорданова супералгебра}{суперкоммутативная алгебра с тождеством Йордана (x²y)x = x²(yx), с суперзнаками}{Суперкоммутативная супералгебра с тождеством Йордана; для чётных $x,y$
\[ (x^2y)\,x=x^2\,(yx), \]
для остальных — со знаками по правилу знаков.}.
  \item Кинг--МакКриммон (1992)\lit{D.~King, K.~McCrimmon, \emph{The Kantor construction of Jordan superalgebras}, Comm. Algebra \textbf{20} (1992), 109--126.}: $K(A)$ йорданова $\iff$ $\{\,,\}$ — \hlt{йорданова скобка}: Лейбниц и Якоби выполнены с поправками через $\{a,1\}$.
  \item Замена $\{a,b\}\mapsto\{a,b\}+\tfrac12\bigl(a\{b,1\}-\{a,1\}b\bigr)$ делает йорданову скобку обобщённой пуассоновой: Якоби точное, поправка только в Лейбнице.
  \item Нечётный вариант — \hlt{обобщённые алгебры Герстенхабера} (Кантарини--Кац 2010\lit{N.~Cantarini, V.~Kac, \emph{Classification of linearly compact simple rigid superalgebras}, IMRN (2010), 3341--3393.}).
\end{itemize}
\end{frame}

\begin{frame}[label=po]{\ft{Примеры с $\D\neq0$: $PO(n,n+1)$}{Примеры с ненулевым D: PO(n,n+1)}}
\small
К $PO(n,n)$ добавим нечётную переменную $\tau$: $PO(n,n+1)=\F[x_1,\dots,x_n\mid\xi_1,\dots,\xi_n,\tau]$ со скобкой ($\{\,,\}_H$ — скобка Баттина, $E$ — оператор Эйлера)
$$\{f,g\}_K=\{f,g\}_H+(E-2)(f)\,\frac{\partial g}{\partial\tau}+(-1)^{|f|}\frac{\partial f}{\partial\tau}\,(E-2)(g),\qquad
E=\sum_{i=1}^n\Bigl(x_i\frac{\partial}{\partial x_i}+\xi_i\frac{\partial}{\partial\xi_i}\Bigr).$$
\begin{itemize}\setlength{\itemsep}{0.15em}

  \item Здесь $\D$ — всё-таки дифференцирование, так как $\{1,1\}=0$. В свободной алгебре будет иначе.
\end{itemize}

Всякая простая линейно компактная обобщённая алгебра Герстенхабера калибровочно эквивалентна $PO(n,n)$ или $PO(n,n+1)$.

\begin{frame}[label=lemma5pf]{\ft{Лемма 5: подробно}{Лемма 5: подробно}}
\small
\begin{thmbox}{Лемма 5}
$U$ порождает $\G$ как в.п.
\end{thmbox}
\textbf{Доказательство.} Гомоморфизм нечётных супералгебр Ли $\psi\colon\LL\to\G$, тождественный на $X$, переводит элементы $M$ в те же элементы $\G$; по \go{setM}{лемме 4} $\psi(\LL)$ натянуто на $M\subset U$.\par\smallskip
Индукцией по $N=\deg_e u+\deg_e v$ докажем, что $\{u,v\}\in\spn U$ для $u,v\in U$.
\bul $u,v\in M$: $\{u,v\}=\psi(\{u,v\})\in\psi(\LL)\subset\spn U$.
\bul $\deg_e v\ge2$: $v=v_1v_2$, $v_1,v_2\in U\setminus\{1\}$, $\deg_e v_1,\deg_e v_2<\deg_e v$. По (5)
\[ \{u,v_1v_2\}=\mka{\{u,v_1\}}\,v_2+(-1)^{\rp u|v_1|}v_1\,\mka{\{u,v_2\}}+(-1)^{\rp u}\mkb{\{1,u\}}\,v_1v_2 ; \]
у пар $(u,v_1)$, $(u,v_2)$, $(1,u)$ сумма $e$-степеней меньше $N$, а $\spn U$ замкнуто относительно умножения.
\bul $\deg_e v\le1$, $\deg_e u\ge2$: $\{u,v\}=-(-1)^{\rp u\rp v}\{v,u\}$ по (3) — предыдущий случай.\par\smallskip
Итак, $\spn U$ — подалгебра $\G$, содержащая $X$, то есть $\spn U=\G$. \hfill$\square$
\look{\leg{hlA}{Жёлтые} скобки — по предположению индукции; \leg{hlB}{голубая} — поправка из (5), её $e$-степень меньше (\go{lemma5bpf}{пример}).}
\backto{lemma5}{к слайду «Порождение: лемма 5»}

\begin{frame}[label=lemma5bpf]{\ft{Лемма 5: почему индукция по $e$-степени}{Лемма 5: почему по e-степени}}

\begin{frame}[label=lemma6pf]{\ft{Лемма 6: формулы (7), (8) корректны}{Лемма 6: корректность}}
\small
\begin{thmbox}{Лемма 6}
Правые части \go{formula8}{(7), (8)} не зависят от разложений $a=a_1\cdots a_m$, $b=b_1\cdots b_n$ на множители $a_i,b_j\in\Lh$. Свойства (i)--(iv) — на \go{lemma6ipf}{отдельных слайдах}.
\end{thmbox}
\textbf{Доказательство.} Правая часть (7) — полилинейная функция $R_m(a_1,\dots,a_m)$, правая часть (8) — $R_{m,n}(a_1,\dots,a_m;\,b_1,\dots,b_n)$. У них два свойства (для $R_{m,n}$ — по каждой группе аргументов):
\[ \begin{aligned}
&\text{(a)}\quad R(\dots,a_{k+1},a_k,\dots)=\mka{(-1)^{|a_k||a_{k+1}|}}\,R(\dots,a_k,a_{k+1},\dots) &&\nm{суперсимметричность, \go{lemma6apf}{проверка}}\\
&\text{(b)}\quad R_m(a_1,\dots,a_{m-1},1)=R_{m-1}(a_1,\dots,a_{m-1}) &&\nm{единица убирается, \go{lemma6bpf}{проверка}}
\end{aligned} \]
Возьмём любое разложение на множители из $\Lh$; по полилинейности можно считать, что множители — из базиса $M$. По (a) они переставляются в порядке базиса, и обе части меняют знак одинаково; если нечётный базисный элемент встретился дважды, обе части равны $0$. По (b) единицы убираются. Остаётся разложение, по которому (7), (8) заданы на $U$.
\look{\leg{hlA}{Жёлтым} — тот же знак, что даёт (2) при перестановке $a_k$ и $a_{k+1}$ в произведении: поэтому $R$ согласуется с умножением в $\Sm$.}

\begin{frame}[label=lemma6apf]{\ft{Лемма 6: суперсимметричность (a)}{Лемма 6: суперсимметричность}}
\small
\textbf{Формула (7).} После перестановки $a_k\leftrightarrow a_{k+1}$ член с $\{1,a_k\}$ стоит на месте $k+1$; перед ним $a_{<k}a_{k+1}$:
\[ \begin{aligned}
&(-1)^{|a_{<k}|+|a_{k+1}|}\,a_{<k}a_{k+1}\{1,a_k\}a_{>k+1}\\
&\qquad=(-1)^{|a_{<k}|+|a_{k+1}|+|a_{k+1}|(|a_k|+1)}\,a_{<k}\{1,a_k\}a_{k+1}a_{>k+1} &&\nm{(2), $|\{1,a_k\}|=|a_k|+1$}\\
&\qquad=\mka{(-1)^{|a_k||a_{k+1}|}}\,(-1)^{|a_{<k}|}\,a_{<k}\{1,a_k\}a_{k+1}a_{>k+1} &&\nm{член $i=k$ до перестановки}
\end{aligned} \]
Члены с $i\neq k,k+1$ и последний член (7) умножаются на тот же множитель: так меняется произведение, а чётность $a_{<i}$ та же.\par\smallskip
\textbf{Формула (8).} Произведения $a$, $a\D(b)$, $a\D(1)b$, $\D(a)b$ умножаются на $(-1)^{|a_k||a_{k+1}|}$; члены двойной суммы с $i\neq k,k+1$ — тоже. Член $i=k$ до перестановки и член $i=k+1$ после:
\[ \begin{aligned}
&(-1)^{|a_k|(|a_{k+1}|+|a_{>k+1}|)}\,a_{<k}a_{k+1}a_{>k+1}\{a_k,b_j\}b_{\hat\jmath},\\
&(-1)^{|a_k||a_{>k+1}|}\,a_{<k}a_{k+1}a_{>k+1}\{a_k,b_j\}b_{\hat\jmath}
\end{aligned} \]
отличаются множителем $\mka{(-1)^{|a_k||a_{k+1}|}}$.
\look{\leg{hlA}{Жёлтым} — множитель из (a): знак $(-1)^{|a_{k+1}|}$ от удлинения $a_{<i}$ и знак переноса $\{1,a_k\}$ через $a_{k+1}$ вместе дают ровно его.}
\backto{axioms\fnov{3}}{к слайду «Аксиомы в модели»}

\begin{frame}[label=lemma6bpf]{\ft{Лемма 6: единица убирается}{Лемма 6: единица убирается}}
\small
Пусть $a_m=1$ и $a'=a_1\cdots a_{m-1}$; тогда $a=a'$ как элемент $\Sm$. Проверяем (b).\par\smallskip
\textbf{Формула (7).} Член с $i=m$ и последний член:
\[ \begin{aligned}
(-1)^{|a'|}\,a'\{1,1\}&=\mkc{\{1,1\}\,a'} &&\nm{$\{1,1\}$ нечётен}\\
-(m-1)\{1,1\}\,a&=-(m-2)\{1,1\}\,a'\;\mkc{-\,\{1,1\}\,a'} &&\nm{$a=a'$}
\end{aligned} \]
Розовые члены сокращаются; остальное — ровно $R_{m-1}(a')$.\par\smallskip
\textbf{Формула (8).} Члены двойной суммы с $i=m$ (здесь $a_{\hat m}=a'$):
\[ \begin{aligned}
\sum_j(-1)^{|b_j||b_{<j}|}\,a'\{1,b_j\}b_{\hat\jmath}&=a'\sum_j(-1)^{|b_{<j}|}\,b_{<j}\{1,b_j\}b_{>j} &&\nm{(2), $|\{1,b_j\}|=|b_j|+1$}\\
&=\mkb{a'\D(b)}+\mkc{(n-1)\,a'\D(1)\,b} &&\nm{по (7)}
\end{aligned} \]
В поправках (8) для $a$ вместо $a'$ коэффициент $m-1$ вместо $m-2$, а $\D(a)=\D(a')$ по уже доказанному, поэтому
\[ \bigl(\text{поправки для }a\bigr)-\bigl(\text{поправки для }a'\bigr)=\mkb{-\,a'\D(b)}\;\mkc{-\,(n-1)\,a'\D(1)\,b}. \]
\look{\leg{hlB}{Голубые} и \leg{hlC}{розовые} члены сокращаются попарно. Случай $b_n=1$ проверяется так же.}

\begin{frame}[label=lemma6iiipf]{\ft{Лемма 6: свойство (iii)}{Лемма 6: свойство (iii)}}

\begin{thmbox}{Лемма 6 (iii)}

\end{aligned} \]

\begin{frame}[label=lemma7pf]{\ft{Лемма 7: кососимметричность $\{\,,\}^*$}{Лемма 7}}
\small
\begin{thmbox}{Лемма 7}
$\{a,b\}^*=-(-1)^{\rp a\rp b}\{b,a\}^*$ для всех $a,b\in\Sm$.
\end{thmbox}
\textbf{Доказательство.} $a=a_1\cdots a_m$, $b=b_1\cdots b_n$; $\{a,b\}_0$ — двойная сумма в (8). Член $(j,i)$ суммы $\{b,a\}_0$:
\[ \begin{aligned}
&(-1)^{|b_j||b_{>j}|+|a_i||a_{<i}|}\,b_{\hat\jmath}\{b_j,a_i\}a_{\hat\imath}\\
&\qquad=-(-1)^{\rp{a_i}\rp{b_j}+|b_j||b_{>j}|+|a_i||a_{<i}|}\,b_{\hat\jmath}\{a_i,b_j\}a_{\hat\imath} &&\nm{(3) в $\Lh$}\\
&\qquad=-(-1)^{\rp a\rp b}\,\mka{(-1)^{|a_i||a_{>i}|+|b_j||b_{<j}|}\,a_{\hat\imath}\{a_i,b_j\}b_{\hat\jmath}} &&\nm{(2), $|\{a_i,b_j\}|=\rp{a_i}+|b_j|$}
\end{aligned} \]
В последнем шаге сравниваются показатели: $\rp a=|a_{\hat\imath}|+\rp{a_i}$, $\rp b=|b_{\hat\jmath}|+\rp{b_j}$, $|a_i|+\rp{a_i}=1$. Значит, $\{b,a\}_0=-(-1)^{\rp a\rp b}\{a,b\}_0$; поправки — на \go{lemma7bpf}{следующем слайде}.
\look{\leg{hlA}{Жёлтым} — член $(i,j)$ суммы $\{a,b\}_0$: члены двух сумм совпадают с точностью до общего знака.}

\begin{frame}[label=lemma7bpf]{\ft{Лемма 7: поправки}{Лемма 7: поправки}}
\small
По (2), так как $|\D(a)|=\rp a$ и $|\D(1)|=1$,
\[ b\D(a)=(-1)^{|b|\rp a}\D(a)b,\qquad \D(b)a=(-1)^{\rp b|a|}a\D(b),\qquad b\D(1)a=-(-1)^{\rp a\rp b}a\D(1)b. \]
Поэтому поправки $\{b,a\}^*$ равны поправкам $\{a,b\}^*$, умноженным на $-(-1)^{\rp a\rp b}$:
\[ \begin{aligned}
&\mkb{-(n-1)\,b\D(a)+(m-1)(-1)^{\rp b}\D(b)\,a}\;\mkc{-(n-1)(m-1)\,b\D(1)a}\\
&\qquad=-(-1)^{\rp a\rp b}\bigl(\mkb{-(m-1)\,a\D(b)+(n-1)(-1)^{\rp a}\D(a)\,b}\;\mkc{-(m-1)(n-1)\,a\D(1)b}\bigr),
\end{aligned} \]
так как $(-1)^{\rp b(|a|+1)}=(-1)^{\rp a\rp b}$ и $(-1)^{|b|\rp a}=(-1)^{\rp a\rp b+\rp a}$. Вместе с двойной суммой это и есть лемма 7. \hfill$\square$

\begin{frame}[label=lemma8pf]{\ft{Лемма 8: правило (5) для $\{\,,\}^*$}{Лемма 8}}
\small
\begin{thmbox}{Лемма 8}
$\{a,bc\}^*=\{a,b\}^*c+(-1)^{\rp a|b|}b\{a,c\}^*+(-1)^{\rp a}\D(a)bc$ для всех $a,b,c\in\Sm$.
\end{thmbox}

так как $|a_{\hat\imath}\{a_i,c_k\}|=\rp a+|c_k|$ и $b$ переносится через него по (2). Итак,
\[ \mka{\{a,bc\}_0=\{a,b\}_0\,c+(-1)^{\rp a|b|}\,b\,\{a,c\}_0}. \]
\look{\leg{hlA}{Жёлтым} — правило (5) без поправки: для двойной суммы оно верно точно, как у Шестакова. Поправки — на \go{lemma8bpf}{следующем слайде}.}

\begin{frame}[label=lemma8bpf]{\ft{Лемма 8: поправки}{Лемма 8: поправки}}
\small
Члены (8) с $\D$; у $bc$ всего $n+p$ множителей. В левой части:
\[ -(m-1)\,a\D(bc)\;\mkb{+(n+p-1)(-1)^{\rp a}\D(a)bc}\;\mkc{-(m-1)(n+p-1)\,a\D(1)bc}. \]

По (2), так как $|\D(a)|=\rp a$ и $|\D(1)|=1$,

\begin{frame}[label=lemma9pf]{\ft{Лемма 9: якобиатор, (i)}{Лемма 9 (i)}}
\small
\begin{thmbox}{Лемма 9}
Пусть $V$ — супералгебра с единицей, в которой выполнены \ax{1}, \ax{2}, \ax{3}, \ax{5}, и $J(a,b,c)=\{a,\{b,c\}\}-\{\{a,b\},c\}-(-1)^{\rp a\rp b}\{b,\{a,c\}\}$. Тогда\\
(i) $J(a,b,c)=-(-1)^{\rp a\rp b}J(b,a,c)=-(-1)^{\rp b\rp c}J(a,c,b)$;\\
(ii) $J(a,b,xy)=J(a,b,x)\,y+(-1)^{(|a|+|b|)|x|}x\,J(a,b,y)-J(a,b,1)\,xy$.
\end{thmbox}
\textbf{(i), первое равенство.} По (3) $\{\{b,a\},c\}=-(-1)^{\rp a\rp b}\{\{a,b\},c\}$, поэтому
\[ -(-1)^{\rp a\rp b}J(b,a,c)=-(-1)^{\rp a\rp b}\{b,\{a,c\}\}\;\mka{-\,\{\{a,b\},c\}}+\{a,\{b,c\}\}=J(a,b,c). \]
\textbf{Второе равенство.} По (3), так как $\rp{\{a,c\}}=\rp a+\rp c$ и $\rp{\{a,b\}}=\rp a+\rp b$:
\[ \begin{aligned}
J(a,c,b)&=\{a,\{c,b\}\}-\{\{a,c\},b\}-(-1)^{\rp a\rp c}\{c,\{a,b\}\}\\
&=-(-1)^{\rp b\rp c}\{a,\{b,c\}\}+(-1)^{\rp b(\rp a+\rp c)}\{b,\{a,c\}\}+(-1)^{\rp a\rp c+\rp c(\rp a+\rp b)}\{\{a,b\},c\}\\
&=\mka{-(-1)^{\rp b\rp c}}\bigl(\{a,\{b,c\}\}-(-1)^{\rp a\rp b}\{b,\{a,c\}\}-\{\{a,b\},c\}\bigr)=-(-1)^{\rp b\rp c}J(a,b,c).
\end{aligned} \]
\look{\leg{hlA}{Жёлтым} — что даёт перестановка аргументов. Доказательство (ii) — \go{lemma9bpf}{далее}.}

\begin{frame}[label=lemma9bpf]{\ft{Лемма 9 (ii): раскрываем по (5)}{Лемма 9 (ii): раскрытие}}
\footnotesize
Раскрываем три члена $J(a,b,xy)$ по (5), используя $|\{b,x\}|=\rp b+|x|$, $|\D(b)|=\rp b$, $\rp{\{a,b\}}=\rp a+\rp b$:
\[ \begin{aligned}
\{a,\{b,xy\}\}={}&\mka{\{a,\{b,x\}\}y}+(-1)^{\rp a(|x|+\rp b)}\{b,x\}\{a,y\}+(-1)^{\rp a}\D(a)\{b,x\}y\\
&+(-1)^{\rp b|x|}\bigl(\{a,x\}\{b,y\}+\mka{(-1)^{\rp a|x|}x\{a,\{b,y\}\}}+(-1)^{\rp a}\D(a)x\{b,y\}\bigr)\\
&+(-1)^{\rp b}\bigl(\mkb{\{a,\D(b)\}xy}+(-1)^{\rp a\rp b}\D(b)\{a,xy\}+(-1)^{\rp a}\D(a)\D(b)xy\bigr),\\
\{\{a,b\},xy\}={}&\mka{\{\{a,b\},x\}y}+\mka{(-1)^{(|a|+|b|)|x|}x\{\{a,b\},y\}}+\mkb{(-1)^{\rp a+\rp b}\D(\{a,b\})xy},\\
\{b,\{a,xy\}\}={}&\mka{\{b,\{a,x\}\}y}+(-1)^{\rp b(|x|+\rp a)}\{a,x\}\{b,y\}+(-1)^{\rp b}\D(b)\{a,x\}y\\
&+(-1)^{\rp a|x|}\bigl(\{b,x\}\{a,y\}+\mka{(-1)^{\rp b|x|}x\{b,\{a,y\}\}}+(-1)^{\rp b}\D(b)x\{a,y\}\bigr)\\
&+(-1)^{\rp a}\bigl(\mkb{\{b,\D(a)\}xy}+(-1)^{\rp a\rp b}\D(a)\{b,xy\}+(-1)^{\rp b}\D(b)\D(a)xy\bigr);
\end{aligned} \]
ещё $\{a,xy\}$ и $\{b,xy\}$ раскрываются по (5). Всё подставляется в
\[ J(a,b,xy)=\{a,\{b,xy\}\}-\{\{a,b\},xy\}-(-1)^{\rp a\rp b}\{b,\{a,xy\}\}. \]
\look{\leg{hlA}{Жёлтые} члены собираются в $J(a,b,x)\,y$ и $x\,J(a,b,y)$, \leg{hlB}{голубые} — в $-J(a,b,1)\,xy$, остальные сокращаются попарно (\go{lemma9cpf}{следующий слайд}).}
\backto{axioms\fnov{3}}{к слайду «Аксиомы в модели»}
\end{frame}

\begin{frame}[label=lemma9cpf]{\ft{Лемма 9 (ii): жёлтые и голубые члены}{Лемма 9 (ii): сборка}}
\small
\bul \leg{hlA}{Жёлтые} члены собираются в якобиаторы:
\[ \begin{aligned}
\{a,\{b,x\}\}y-\{\{a,b\},x\}y-(-1)^{\rp a\rp b}\{b,\{a,x\}\}y&=\mka{J(a,b,x)\,y},\\
(-1)^{(|a|+|b|)|x|}\,x\bigl(\{a,\{b,y\}\}-\{\{a,b\},y\}-(-1)^{\rp a\rp b}\{b,\{a,y\}\}\bigr)&=\mka{(-1)^{(|a|+|b|)|x|}\,x\,J(a,b,y)}.
\end{aligned} \]
\bul \leg{hlB}{Голубые} члены. По (3)
\[ \{b,1\}=-(-1)^{\rp b}\D(b),\qquad \{a,1\}=-(-1)^{\rp a}\D(a),\qquad \{\{a,b\},1\}=-(-1)^{\rp a+\rp b}\D(\{a,b\}), \]
поэтому
\[ J(a,b,1)=-(-1)^{\rp b}\{a,\D(b)\}+(-1)^{\rp a+\rp b}\D(\{a,b\})+(-1)^{\rp a\rp b+\rp a}\{b,\D(a)\}, \]
и голубые члены $J(a,b,xy)$ в сумме дают $\mkb{-\,J(a,b,1)\,xy}$.\par\smallskip
Остальные шестнадцать членов сокращаются попарно — \go{lemma9dpf}{следующий слайд}.
\backto{axioms\fnov{3}}{к слайду «Аксиомы в модели»}
\end{frame}

\begin{frame}[label=lemma9dpf]{\ft{Лемма 9 (ii): остальные члены сокращаются}{Лемма 9 (ii): сокращение}}
\small
После раскрытия $\{a,xy\}$ и $\{b,xy\}$ по (5) остальные шестнадцать членов $J(a,b,xy)$ собираются в восемь пар:
\[ \begin{aligned}
&\bigl((-1)^{\rp a(|x|+\rp b)}-(-1)^{\rp a\rp b+\rp a|x|}\bigr)\,\{b,x\}\{a,y\}\\
&\bigl((-1)^{\rp b|x|}-(-1)^{\rp a\rp b+\rp b(|x|+\rp a)}\bigr)\,\{a,x\}\{b,y\}\\
&\bigl((-1)^{\rp a}-(-1)^{\rp a}\bigr)\,\D(a)\{b,x\}y\\
&\bigl((-1)^{\rp b|x|+\rp a}-(-1)^{\rp a+\rp b|x|}\bigr)\,\D(a)x\{b,y\}\\
&\bigl((-1)^{\rp b+\rp a\rp b}-(-1)^{\rp a\rp b+\rp b}\bigr)\,\D(b)\{a,x\}y\\
&\bigl((-1)^{\rp b+\rp a\rp b+\rp a|x|}-(-1)^{\rp a\rp b+\rp a|x|+\rp b}\bigr)\,\D(b)x\{a,y\}\\
&\bigl((-1)^{\rp b+\rp a}-(-1)^{\rp a+\rp b}\bigr)\,\D(a)\D(b)xy\\
&\bigl((-1)^{\rp b+\rp a\rp b+\rp a}-(-1)^{\rp a\rp b+\rp a+\rp b}\bigr)\,\D(b)\D(a)xy
\end{aligned} \]
В каждой паре показатели равны по модулю $2$, так что все восемь коэффициентов — нули. Итак,
\[ J(a,b,xy)=\mka{J(a,b,x)\,y+(-1)^{(|a|+|b|)|x|}x\,J(a,b,y)}\;\mkb{-\,J(a,b,1)\,xy}. \qquad\square \]
\backto{axioms\fnov{3}}{к слайду «Аксиомы в модели»}
\end{frame}

\begin{frame}[label=prop10pf]{\ft{Предложение 10: подробно}{Предложение 10: подробно}}
\small
\begin{thmbox}{Предложение 10}
$J=0$ на $\Sm$, то есть в $\Sm$ выполнено тождество Якоби (4).
\end{thmbox}
По \go{lemma6pf}{леммам 6}--\go{lemma8pf}{8} к $\Sm$ применима \go{lemma9pf}{лемма 9}.\par\smallskip
\textbf{Шаг 1.} Фиксируем $a,b\in\Sm$. Если $J(a,b,u)=0$ при всех $u\in\Lh$, то $J(a,b,c)=0$ при всех $c\in\Sm$. Действительно, $\Sm$ натянуто на произведения $c_1\cdots c_m$, $c_k\in\Lh$, и по лемме 9 (ii)
\[ \begin{aligned}
J(a,b,c_1\cdots c_m)={}&\mka{J(a,b,c_1\cdots c_{m-1})}\,c_m+(-1)^{(|a|+|b|)|c_1\cdots c_{m-1}|}\,c_1\cdots c_{m-1}\,\mka{J(a,b,c_m)}\\
&\mkb{-\,J(a,b,1)}\,c_1\cdots c_m;
\end{aligned} \]
индукция по $m$, так как $1,c_m\in\Lh$.
\look{\leg{hlA}{Жёлтые} — нули по предположению индукции, \leg{hlB}{голубой} — нуль, так как $1\in\Lh$. Шаги 2--4 — \go{prop10bpf}{на следующем слайде}.}
\backto{jacobim}{к слайду «Тождество Якоби в модели»}
\end{frame}

\begin{frame}[label=prop10bpf]{\ft{Предложение 10: подробно, шаги 2--4}{Предложение 10: шаги 2--4}}
\small
\textbf{Шаг 2.} $u,v,w\in\Lh$: $J(u,v,w)=0$, так как по \go{lemma6ipf}{лемме 6 (i)} это якобиатор $\LL$. По шагу 1 (с $a=u$, $b=v$)
\[ J(u,v,c)=0\qquad\text{для всех}\ c\in\Sm. \]
\textbf{Шаг 3.} По лемме 9 (i) и шагу 2
\[ J(u,c,v)=-(-1)^{\rp c\rp v}J(u,v,c)=0\qquad(u,v\in\Lh,\ c\in\Sm), \]
и по шагу 1 (с $a=u$, $b=c$) $J(u,b,c)=0$ для всех $u\in\Lh$, $b,c\in\Sm$.\par\smallskip
\textbf{Шаг 4.} По лемме 9 (i) и шагу 3
\[ J(a,b,u)=-(-1)^{\rp b\rp u}J(a,u,b)=(-1)^{\rp b\rp u+\rp a\rp u}\,\mka{J(u,a,b)}=0\qquad(a,b\in\Sm,\ u\in\Lh), \]
и по шагу 1 $J(a,b,c)=0$ для всех $a,b,c\in\Sm$. \hfill$\square$
\look{\leg{hlA}{Жёлтым} — якобиатор с первым аргументом из $\Lh$: он нуль по шагу 3.}
\backto{jacobim}{к слайду «Тождество Якоби в модели»}
\end{frame}

\begin{frame}[label=shestakovpf]{Теорема Шестакова: доказательство}
\begin{sidecols}
\sidecol[0.2]
\begin{sidebox}{formulas1}
\sromn{1.2}{&\{a,b\}=-\{b,a\}}
\sromn{1.3}{&\{a,bc\}=\{a,b\}\,c\\&\;+b\,\{a,c\}}
\end{sidebox}
\maincol[0.78]\vspace*{-6pt}
$P(X)$ и $\mathfrak g=\mathrm{Lie}(X)$ — \go{freealg}{свободные} алгебры Пуассона и Ли на $X$.

\medskip
\textbf{1. Порождение.} По \fn{1.2}, \fn{1.3} скобка с произведением раскладывается в сумму произведений скобок от букв; скобки от букв по \go{good}{теореме Холла} — комбинации хороших слов:
\[ \{x_1,x_2^2\}=2x_2\{x_1,x_2\}=-2\,x_2\{x_2,x_1\} \qquad\nm{по \fn{1.3}, \fn{1.2}} \]
Значит, одночлены от хороших слов порождают $P(X)$.

\onslide<2->
\medskip
\textbf{2. Независимость.} $S(\mathfrak g)$ со \go{poisson\fnov{4}}{скобкой Ли--Пуассона} — алгебра Пуассона, поэтому по \go{freealg}{определению свободной алгебры} есть гомоморфизм $\varphi\colon P(X)\to S(\mathfrak g)$, тождественный на $X$:
\[ \begin{aligned}
\varphi(u)&=u\in\mathfrak g &&\nm{$u$ — хорошее слово; на $\mathfrak g$ скобка в $S(\mathfrak g)$ — скобка $\mathfrak g$}\\
\varphi(u_1\cdots u_r)&=u_1\cdots u_r &&\nm{$\varphi$ сохраняет произведение}
\end{aligned} \]
Хорошие слова — базис $\mathfrak g$ (\go{good}{Холл}), а одночлены от базиса $\mathfrak g$ — базис $S(\mathfrak g)$. Поэтому $\varphi$ переводит одночлены от хороших слов в попарно различные элементы базиса $S(\mathfrak g)$, и в $P(X)$ они линейно независимы.
\end{sidecols}
