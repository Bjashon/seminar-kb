\begin{frame}[label=poisson]{Алгебры Пуассона}
\begin{defbox}{Алгебра Пуассона}
Коммутативная ассоциативная алгебра $A$ с $1$ и скобкой Ли $\{\,,\}$, связанные \hlt{правилом Лейбница}
$\{a,bc\}=\{a,b\}\,c+b\,\{a,c\}$: при каждом $a$ отображение $\{a,\cdot\}$ — дифференцирование $A$.
\end{defbox}
\begin{exbox}{Пример: $\F[p,q]$ (механика), \ $\{f,g\}=\frac{\partial f}{\partial p}\frac{\partial g}{\partial q}-\frac{\partial f}{\partial q}\frac{\partial g}{\partial p}$, \ $\{p,q\}=1$}
$$\{p,q^2\}=1\cdot2q-0\cdot0=2q,\qquad \{p,q\}\,q+q\,\{p,q\}=q+q=2q .$$
\end{exbox}
\begin{itemize}\setlength{\itemsep}{0.25em}
  \item \hlt{Единицу скобка не видит:} \ $\{a,1\}=\{a,1\cdot1\}=2\{a,1\}$, \ значит $\{a,1\}=0$.
  \item Скобку достаточно знать на образующих. На $S(\mathfrak g)$ так получается \hlt{скобка Ли--Пуассона}.
\end{itemize}
\end{frame}

\begin{frame}[label=super]{Супералгебры: правило знаков}
\begin{itemize}\setlength{\itemsep}{0.3em}
  \item $\F$ — поле характеристики $0$. Пространства $\Z_2$-градуированы: $V=V_0\oplus V_1$, \ $|a|\in\Z_2$ — \tip{чётность}{0 — чётный элемент, 1 — нечётный} однородного $a$.
  \item \hlt{Правило знаков:} переставили два нечётных соседа — множитель $-1$. Суперкоммутативность $ab=(-1)^{|a||b|}ba$; для нечётного $a$: \ $a^2=0$.
  \item \hlt{Супердифференцирование} чётности $p$: \ $d(ab)=d(a)\,b+(-1)^{p|a|}\,a\,d(b)$.
\end{itemize}
\begin{exbox}{Пример: суперполиномы $\F[x\mid\xi_1,\xi_2]$, \ $x$ чётная, \ $\xi_1,\xi_2$ нечётные}
Базис: $x^k,\ x^k\xi_1,\ x^k\xi_2,\ x^k\xi_1\xi_2$ — так как $\xi_2\xi_1=-\xi_1\xi_2$ и $\xi_i^2=0$. Производная $\partial/\partial\xi_1$ нечётна:
$$\frac{\partial}{\partial\xi_1}(\xi_2\xi_1)=-\frac{\partial}{\partial\xi_1}(\xi_1\xi_2)=-\xi_2 .$$
\end{exbox}
{\small Супералгебра Ли: \ $[a,b]=-(-1)^{|a||b|}[b,a]$, \quad $[a,[b,c]]=[[a,b],c]+(-1)^{|a||b|}[b,[a,c]]$.}
\end{frame}

\begin{frame}[label=gerst]{Алгебры Герстенхабера: нечётная скобка}
Скобка \hlt{нечётна}: $|\{a,b\}|=|a|+|b|+1$. \ \hlt{Обращённая чётность}: $\rp a=|a|+1$.
\begin{defbox}{Алгебра Герстенхабера}
\small Суперкоммутативная ассоциативная супералгебра с $1$ и нечётной скобкой, которая — \tip{супералгебра Ли}{то же пространство с чётностью ‖a‖ = |a| + 1: скобка в нём чётная} относительно чётности $\rp{\cdot}$:
$$\begin{gathered}
\{a,b\}=-(-1)^{\rp a\rp b}\{b,a\},\qquad \{a,\{b,c\}\}=\{\{a,b\},c\}+(-1)^{\rp a\rp b}\{b,\{a,c\}\},\\
\{a,bc\}=\{a,b\}c+(-1)^{\rp a|b|}b\{a,c\}.
\end{gathered}$$
\end{defbox}
\begin{exbox}{Пример: $\Lambda(\mathfrak g)$, скобка Схоутена; \ $\mathfrak g=\langle h,e\rangle$, $[h,e]=2e$; элементы $\mathfrak g$ нечётны}
\small $\{h,he\}=\{h,h\}e+h\{h,e\}=0+2he$, \qquad $\{e,he\}=\{e,h\}e+h\{e,e\}=-2e^2+0=0$.
\end{exbox}
{\small\textbf{Отличие от Пуассона:} для чётных $a,b$ скобка \hlt{симметрична}, и $\{a,a\}$ может быть $\neq0$.}
\end{frame}

\begin{frame}[label=genpoisson]{Правило Лейбница с поправкой}
В алгебре Пуассона $\{a,1\}=0$. Что будет, если разрешить $\{a,1\}\neq0$?
\begin{defbox}{Обобщённая алгебра Пуассона (скобка чётная; Кантарини--Кац 2007)}
Скобка Ли и
$$\{a,bc\}=\{a,b\}c+b\{a,c\}\;\mka{-\,\{a,1\}\,bc}\,:$$
$\{a,\cdot\}$ — дифференцирование плюс умножение на элемент.
\end{defbox}
\begin{exbox}{Пример: $\F[x]$, \ $\{f,g\}=fg'-f'g$ — скобка векторных полей $f\,\tfrac{d}{dx}$}
$\{f,1\}=-f'\neq0$. Проверим правило на $\{x,x^2\}$:
$$\{x,x^2\}=x\cdot2x-1\cdot x^2=x^2,\qquad \{x,x\}\,x+x\,\{x,x\}-\{x,1\}\,x^2=0+0+x^2 .$$
\end{exbox}
{\small\color{mute}Откуда поправка: йордановы скобки и \go{kantor}{двойник Кантора}.}
\end{frame}

\begin{frame}[label=def1]{Обобщённые алгебры Герстенхабера}
\begin{defbox}{Определение 1}
\tip{$\Z_2$-градуированное}{V = V₀ ⊕ V₁; |a| — чётность однородного a; ‖a‖ = |a| + 1} пространство $V$ с $1\in V_0$ и операциями $ab$, $\{a,b\}$:
$$\begin{aligned}
&(1)\quad |ab|=|a|+|b|,\qquad |\{a,b\}|=|a|+|b|+1;\\
&(2)\quad (ab)c=a(bc),\qquad ab=(-1)^{|a||b|}ba,\qquad 1a=a;\\
&(3)\quad \{a,b\}=-(-1)^{\rp a\rp b}\{b,a\};\\
&(4)\quad \{a,\{b,c\}\}=\{\{a,b\},c\}+(-1)^{\rp a\rp b}\{b,\{a,c\}\};\\
&(5)\quad \{a,bc\}=\{a,b\}c+(-1)^{\rp a|b|}b\{a,c\}+\mka{(-1)^{\rp a}\D(a)\,bc},\qquad \D(a)=\{1,a\}.
\end{aligned}$$
\end{defbox}
\vskip-0.1cm
\begin{itemize}\small\setlength{\itemsep}{0em}
  \item Поправка та же, что в чётном случае: по (3) $(-1)^{\rp a}\D(a)=-\{a,1\}$. При $\D=0$ — алгебры Герстенхабера.
  \item Odd generalized Poisson superalgebras Кантарини--Каца (2010); простые — \go{po}{$PO(n,n)$, $PO(n,n+1)$}.
\end{itemize}
\end{frame}

\begin{frame}[label=ex1]{\ft{Новое явление: элемент $\{1,1\}$}{Новое явление: элемент {1,1}}}
\begin{itemize}\setlength{\itemsep}{0.3em}
  \item \textbf{Чётная скобка:} $\{1,1\}=-\{1,1\}$, значит $\{1,1\}=0$.
  \item \textbf{Нечётная:} $\rp1=1$, и \ax{3} при $a=b=1$ даёт $\{1,1\}=-(-1)^{1\cdot1}\{1,1\}=\{1,1\}$ — условия \hlt{нет}.
  \item \ax{5} при $a=1$: \ $\D(bc)=\D(b)\,c+(-1)^{|b|}b\,\D(c)\;\mka{-\,\D(1)\,bc}$. \ $\D$ — дифференцирование $\iff$ $\{1,1\}=0$.
\end{itemize}
\begin{exbox}{Пример 1: \ $V=\F1\oplus\F\varepsilon$, \ $|\varepsilon|=1$, \ $\varepsilon^2=0$; \ $\{1,1\}=\varepsilon$, остальные скобки — нули}
\ax{3}, \ax{4}: все скобки лежат в $\F\varepsilon$, а $\{\varepsilon,V\}=0$. \quad \ax{5} при $a=b=c=1$: \ $\varepsilon=\varepsilon+\varepsilon-\varepsilon$.
$$\D(1\cdot1)=\varepsilon\ \neq\ 2\varepsilon=\D(1)\,1+1\,\D(1).$$
\end{exbox}
\textbf{Аксиомы не заставляют $\{1,1\}=0$.} В свободной алгебре $\{1,1\}\neq0$ — это \go{cor13}{следствие 13}.
\end{frame}

\partslide{02}{Задача: базис\\ свободной алгебры}

\begin{frame}[label=freealg]{Свободная алгебра и её базис}
\begin{itemize}\setlength{\itemsep}{0.3em}
  \item \hlt{Свободная алгебра} класса на буквах $X$: выражения от букв, отождествлённые только по тождествам класса. Любое отображение букв в алгебру класса продолжается до гомоморфизма.
  \item \hlt{Базис} свободной алгебры — нормальная форма: равенство элементов проверяется сравнением записей.
  \item \hlt{Мультилинейная часть}: элементы, где каждая из букв $x_1,\dots,x_n$ входит ровно один раз. Её размерность — числовой «паспорт» класса.
\end{itemize}
\vskip0.05cm
{\small\renewcommand{\arraystretch}{1.3}
\begin{tabular}{@{}L{3.2cm}L{6.8cm}cc@{}}
\toprule
\textbf{Класс} & \textbf{мультилинейные элементы от $x_1,x_2,x_3$} & \textbf{$n=3$} & \textbf{любое $n$}\\
\midrule
коммутативные & $x_1x_2x_3$ & $1$ & $1$\\
ассоциативные & $x_1x_2x_3,\ x_1x_3x_2,\ x_2x_1x_3,\ \dots,\ x_3x_2x_1$ & $6$ & $n!$\\
алгебры Ли & $[[x_2,x_1],x_3],\ \ [[x_3,x_1],x_2]$ & $2$ & $(n-1)!$\\
алгебры Пуассона & $x_1x_2x_3,\ \{x_2,x_1\}x_3,\ \dots,\ \{\{x_3,x_1\},x_2\}$ & $6$ & $n!$\\
\bottomrule
\end{tabular}}
\end{frame}

\begin{frame}[label=good]{Алгебры Ли: хорошие слова Холла}
\begin{defbox}{Хорошие слова (алфавит линейно упорядочен)}
\small Буквы — хорошие слова. Слово длины $>1$ \hlt{хорошее}, если оно равно $[u,v]$, где $u,v$ хорошие, $u>v$, и при $u=[u_1,u_2]$ выполнено $u_2\le v$.
Порядок: длиннее — больше; при равной длине — лексикографически по паре $(u,v)$.
\end{defbox}
\begin{exbox}{Пример: $x_1<x_2<x_3$, каждая буква по разу}
\small
Длина 2: \ $[x_2,x_1]<[x_3,x_1]<[x_3,x_2]$. \quad Длина 3: \ $[[x_2,x_1],x_3]$ и $[[x_3,x_1],x_2]$ — хорошие;\\[0.15em]
$[[x_3,x_2],x_1]$ — нет, так как $x_2>x_1$; по Якоби оно равно $[[x_3,x_1],x_2]-[[x_2,x_1],x_3]$.
\end{exbox}
\begin{thmbox}{Теорема (М.~Холл, 1950)}
\small Хорошие слова — базис свободной алгебры Ли; мультилинейных на $k$ буквах — $(k-1)!$.
\end{thmbox}
\end{frame}

\begin{frame}[label=superlie]{Супералгебры Ли: квадраты нечётных}
Для нечётного $v$: \ $[v,v]=-(-1)^{1\cdot1}[v,v]=[v,v]$ — квадрат \hlt{не обязан} быть нулём.
\begin{exbox}{Пример: $\mathfrak{gl}(1|1)$, \ $v=E_{12}+E_{21}$ нечётен, \ для нечётных $[a,b]=ab+ba$}
$$[v,v]=2v^2=2(E_{12}E_{21}+E_{21}E_{12})=2(E_{11}+E_{22})\neq0,\qquad [v,[v,v]]=[v,2E]=0 .$$
\end{exbox}
\begin{itemize}\small\setlength{\itemsep}{0.15em}
  \item Всегда $[v,[v,v]]=0$ и $[[v,v],w]=2[v,[v,w]]$ (Якоби): квадраты нужны только как отдельные элементы базиса.
  \item Свободная супералгебра Ли на одной нечётной букве $\xi$ двумерна: базис $\xi,\ [\xi,\xi]$.
\end{itemize}
\vskip-0.1cm
\begin{thmbox}{Теорема (А.\,С.~Штерн, 1986)}
\small Базис свободной супералгебры Ли: хорошие слова и \hlt{квадраты $[v,v]$ нечётных хороших слов $v$}.
\end{thmbox}
\end{frame}

\begin{frame}[label=shestakov]{Алгебры Пуассона: шаблон Шестакова}
\begin{thmbox}{Теорема (И.\,П.~Шестаков, 1993)}
\small Свободная алгебра Пуассона на $X$ — это $S(\mathrm{Lie}(X))$: базис — \hlt{одночлены от хороших слов}.
\end{thmbox}
\begin{exbox}{Пример: мультилинейные элементы от $x_1<x_2<x_3$; \ всего $1+3+2=6=3!$}
$$\underbrace{x_1x_2x_3}_{1},\quad
\underbrace{\{x_2,x_1\}x_3,\ \{x_3,x_1\}x_2,\ \{x_3,x_2\}x_1}_{3},\quad
\underbrace{\{\{x_2,x_1\},x_3\},\ \{\{x_3,x_1\},x_2\}}_{2}$$
\end{exbox}
\begin{itemize}\small\setlength{\itemsep}{0.15em}
  \item \hlt{Порождение}: Лейбниц выносит произведения из скобок, $\{x_1,x_2^2\}=-2x_2\{x_2,x_1\}$. \hlt{Независимость}: модель $S(\mathrm{Lie}(X))$.
  \item \hlt{Почему $n!$}: блоки букв и хорошее слово на каждом; слов на $k$ буквах $(k-1)!$, как циклов, — итого перестановка.
\end{itemize}
\end{frame}

\begin{frame}[label=letter1]{\ft{Когда $\{a,1\}\neq0$, единица — буква}{Когда {a,1} не нуль, единица — буква}}
\begin{exbox}{Пример: $\{x_1,x_2x_3\}$ по \ax{5}; буквы чётные, $\rp{x_i}=1$}
$$\begin{aligned}
\{x_1,x_2x_3\}&=\{x_1,x_2\}x_3+x_2\{x_1,x_3\}-\{1,x_1\}x_2x_3\\
&=x_3\{x_2,x_1\}+x_2\{x_3,x_1\}\;\mka{-\,x_2x_3\{x_1,1\}} .
\end{aligned}$$
Слева буквы $1$ нет, а в ответе появилось слово $\{x_1,1\}$.
\end{exbox}
\begin{itemize}\setlength{\itemsep}{0.35em}
  \item Значит, $1$ — \hlt{буква скобки}: $X=\{1<x_1<x_2<\cdots\}$. Новые элементы: $\{x_1,1\}$, $\{\{x_1,1\},1\}$, $\{1,1\}$, \dots
  \item Чётный случай (Кайгородов 2017): базис — одночлены от базиса свободной супералгебры Ли на $X$, без множителя $1$; мультилинейная часть по $1,x_1,\dots,x_n$ — $n\cdot n!$.
\end{itemize}
\end{frame}

\footnotesize\renewcommand{\arraystretch}{1.4}
\begin{tabular}{@{}L{4.2cm}L{5.4cm}L{2.8cm}@{}}
\toprule
\textbf{Класс} & \textbf{Базис свободной алгебры} & \textbf{Кто}\\
\midrule
алгебры Ли & хорошие слова & Холл 1950\\
супералгебры Ли & хорошие слова и квадраты нечётных & Штерн 1986\\

обобщённые алгебры Пуассона & то же, буква $1$ в скобках & Кайгородов 2017\\
\rowcolor{softgreen}
\textbf{обобщённые алгебры Герстенхабера} & \textbf{?} & \textbf{эта работа}\\
\bottomrule
\end{tabular}
\normalsize
\vskip0.35cm
\begin{thmbox}{Вопрос}
Найти базис свободной унитальной обобщённой алгебры Герстенхабера $\G$ на $X=\{1<x_1<x_2<\cdots\}$ и размерность её мультилинейной части. Что изменят нечётная скобка и элемент $\{1,1\}$?
\end{thmbox}
\end{frame}

% ================================================================ 3. ответ
\partslide{03}{Ответ}

\begin{frame}[label=setM]{\ft{Скобочная часть: множество $M$}{Скобочная часть: множество M}}
$$M=\bigl\{\,u,\ \{v,v\}\ \bigm|\ u,v\ \text{— хорошие слова над}\ X,\ \ |v|=0\,\bigr\},\qquad e_1=1<e_2<e_3<\cdots$$
\vskip-0.2cm
\begin{thmbox}{Лемма 4}
$M$ — базис свободной \tip{нечётной супералгебры Ли}{кососимметричность (3) и Якоби (4); то же, что супералгебра Ли с чётностью ‖a‖} $\LL$ на $X$.\\
{\small Теорема Штерна для чётности $\rp{\cdot}$: квадраты — для \hlt{чётных} $v$, так что $\{1,1\}\in M$.}
\end{thmbox}
\begin{exbox}{Пример: $X=\{1<x_1\}$, \ $x_1$ чётная}
\small\raggedright
Степень 1: \ $1,\ x_1$. \qquad Степень 2: \ $\{1,1\}<\{x_1,1\}<\{x_1,x_1\}$.\\
Степень 3: \ $\{\{x_1,1\},1\},\ \ \{\{x_1,1\},x_1\}$.\\[0.25em]
Не новые: \ $\{1,x_1\}=\{x_1,1\}$ — скобка чётных симметрична; \ $\{\{1,1\},x_1\}=2\{1,\{1,x_1\}\}=-2\{\{x_1,1\},1\}$ — Якоби.
\end{exbox}
\end{frame}

\begin{frame}[label=thm12]{Главная теорема}
$$U=\bigl\{\,e_{i_1}^{k_1}\cdots e_{i_r}^{k_r}\ \bigm|\ 1<i_1<\dots<i_r,\ k_j\ge1,\ k_j=1\ \text{при}\ |e_{i_j}|=1\,\bigr\}$$
\vskip-0.15cm
\centerline{\small\color{mute}суперкоммутативные одночлены от $M\setminus\{1\}$; пустой одночлен — это $1$}
\vskip0.1cm
\begin{thmbox}{Теорема 12}
$U$ — базис свободной унитальной обобщённой алгебры Герстенхабера $\G$ на $X$. Более того, $\G\cong S(\LL')$, \ $\LL'=\spn(M\setminus\{1\})$, с явно заданной скобкой.
\end{thmbox}
\begin{exbox}{Пример: $X=\{1<x_1\}$, \ $x_1$ чётная}
\small\raggedright
В $U$: \ $1$; \ $x_1,\ \{1,1\},\ \{x_1,1\},\ \{x_1,x_1\}$; \ $x_1^2,\ x_1\{1,1\},\ x_1\{x_1,1\},\ \{1,1\}\{x_1,1\},\ \dots$\\[0.15em]
Не в $U$: \ $\{1,1\}^2=0$ и $\{x_1,1\}^2=0$ — эти элементы нечётны; \ множитель $1$.\\[0.15em]
Разложение по \ax{5}: \ $\{x_1,x_1^2\}=\{x_1,x_1\}x_1+x_1\{x_1,x_1\}-\{1,x_1\}x_1^2=2x_1\{x_1,x_1\}-x_1^2\{x_1,1\}$.
\end{exbox}
\end{frame}

\begin{frame}[label=cor13]{\ft{Следствия: $\{1,1\}\neq0$ и первые размерности}{Следствия}}
\begin{thmbox}{Следствие 13}
В $\G$: \ $\{1,1\}\neq0$, и $\D$ — не дифференцирование.
Более того, $\{1,1\}$, \ $\{1,1\}x_i$, \ $\{1,x_i\}$ линейно независимы.
\end{thmbox}
{\small\color{mute}Можно и без теоремы 12: гомоморфизм $\G\to V$ в \go{ex1}{пример 1}, $1\mapsto1$, $x_i\mapsto0$, переводит $\{1,1\}$ в $\varepsilon\neq0$.}
\vskip0.25cm
\begin{exbox}{Пример 2: элементы $U$, где каждая из букв $1,x_1,\dots,x_n$ входит ровно раз}
$n=1$: \quad $\{x_1,1\}$ \hfill $1=1\cdot1!$\\[0.4em]
$n=2$: \quad $\{x_1,1\}\,x_2,\quad \{x_2,1\}\,x_1,\quad \{\{x_1,1\},x_2\},\quad \{\{x_2,1\},x_1\}$ \hfill $4=2\cdot2!$\\[0.4em]
{\small\color{mute}Не считаются: $\{1,1\}$ — буква $1$ дважды; \ $x_1x_2$ и $\{x_2,x_1\}$ — буквы $1$ нет.}
\end{exbox}
\end{frame}

\begin{frame}[label=lemma5]{\ft{Порождение: $U$ порождает $\G$}{Порождение: U порождает G}}
\begin{itemize}\setlength{\itemsep}{0.35em}
  \item Скобки элементов $M$ лежат в $\spn M$: в $\G$ выполнены (3) и (4), а $M$ — базис $\LL$ (\go{setM}{лемма 4}).
  \item Скобку с произведением раскладываем по \ax{5}:
  $$\{u,v_1v_2\}=\{u,v_1\}v_2+(-1)^{\rp u|v_1|}v_1\{u,v_2\}+(-1)^{\rp u}\{1,u\}\,v_1v_2 .$$
  \item Индукция по сумме \tip{$e$-степеней}{e-степень одночлена из U — число его множителей из M без 1; у единицы она 0} аргументов: справа она у каждой скобки меньше.
\end{itemize}
\begin{exbox}{Почему не индукция по числу букв: \ $\{x_1,x_2x_3\}$ \go{letter1}{из части 2}}
В $\{x_1,x_2x_3\}$ три буквы, а в поправке $-x_2x_3\{x_1,1\}$ — четыре: правило \ax{5} добавляет букву $1$.\\[0.2em]
Сумма $e$-степеней убывает: \ $1+2=3$ у пары $(x_1,\,x_2x_3)$; \ $2$ у пар $(x_1,x_2)$, $(x_1,x_3)$; \ $1$ у пары $(1,x_1)$.
\end{exbox}
\end{frame}

\begin{frame}[label=model]{\ft{Модель: скобка на $S(\LL')$ шаг за шагом}{Модель: скобка на S(L') шаг за шагом}}
\small
$\LL'=\spn(M\setminus\{1\})$, \ $\Sm=S(\LL')$ — суперполиномы, \ $\Lh=\F1\oplus\LL'\cong\LL$: буква $1$ — это единица $\Sm$. \
Нужна скобка на $\Sm$, продолжающая скобку $\Lh$, с аксиомами (1)--(5).
\begin{enumerate}\setlength{\itemsep}{0.45em}
\item \hlt{Как у Шестакова} ($\D=0$): «каждый множитель с каждым»,
$$\{a_1\cdots a_m,\;b_1\cdots b_n\}=\sum_{i,j}\pm\,a_{\hat\imath}\,\{a_i,b_j\}\,b_{\hat\jmath},\qquad a_{\hat\imath}\ \text{— произведение без}\ a_i .$$
\item \hlt{Каждое применение \ax{5} добавляет поправку.} Для $a\in\Lh$:
$$\{a,b_1b_2b_3\}=\{a,b_1\}b_2b_3\pm b_1\{a,b_2\}b_3\pm b_1b_2\{a,b_3\}\;\mkb{-\,2\,\{a,1\}\,b_1b_2b_3}\,:$$
$n$ множителей — $n-1$ поправок.
\item \hlt{Произведения с обеих сторон}: поправки по каждому аргументу и \hlt{поправка к поправке} с $\D(1)=\{1,1\}$.
\end{enumerate}
\end{frame}

\begin{frame}[label=formula8]{Формулы (7) и (8)}
\small
$a=a_1\cdots a_m$, \ $b=b_1\cdots b_n$, \ $a_i,b_j\in\Lh$; \ \ $a_{<i}=a_1\cdots a_{i-1}$, \ $a_{>i}=a_{i+1}\cdots a_m$, \ $a_{\hat\imath}=a_{<i}a_{>i}$.
\begin{defbox}{Формулы (7), (8)}
$$\D(a)=\sum_{i=1}^m(-1)^{|a_{<i}|}\,a_{<i}\{1,a_i\}a_{>i}\;-\;(m-1)\{1,1\}\,a,$$
$$\begin{aligned}
\{a,b\}^*={}&\mka{\sum_{i,j}(-1)^{|a_i||a_{>i}|+|b_j||b_{<j}|}\,a_{\hat\imath}\{a_i,b_j\}b_{\hat\jmath}}\\
&\mkb{-(m-1)\,a\D(b)+(n-1)(-1)^{\rp a}\D(a)\,b}\;\mkc{-(m-1)(n-1)\,a\D(1)\,b}\,.
\end{aligned}$$
\end{defbox}
\leg{hlA}{Жёлтым} — «каждый с каждым», как у Шестакова. \ \leg{hlB}{Синим} — поправки от \ax{5} по каждому аргументу. \ \leg{hlC}{Розовым} — поправка к поправке; её нет, если $\{1,1\}=0$.\\[0.35em]
\textbf{Проверка.} При $m=1$, $n=2$ формула (8) — это ровно \ax{5}. При $m=2$ формула (7) — это $\D(bc)=\D(b)c+(-1)^{|b|}b\D(c)-\D(1)bc$ \go{ex1}{из части 1}.
\end{frame}

\begin{frame}[label=axioms]{Аксиомы в модели: якобиатор}
\begin{itemize}\small\setlength{\itemsep}{0.2em}
  \item \go{props}{Леммы 6--8}: формулы корректны, а \ax{3} и \ax{5} проверяются прямо — формула замкнутая, без рекурсии.
  \item Трудное — тождество Якоби. \tip{Якобиатор}{J = 0 на всех тройках ⇔ выполнено тождество Якоби}: \ $J(a,b,c)=\{a,\{b,c\}\}-\{\{a,b\},c\}-(-1)^{\rp a\rp b}\{b,\{a,c\}\}$.
\end{itemize}
\begin{thmbox}{Лемма 9 (в любой алгебре с аксиомами (1), (2), (3), (5), без Якоби)}
$J$ суперкососимметричен по любой паре аргументов, и
$$J(a,b,xy)=J(a,b,x)\,y+(-1)^{(|a|+|b|)|x|}\,x\,J(a,b,y)\;\mkc{-\,J(a,b,1)\,xy}.$$
\end{thmbox}
\begin{itemize}\small\setlength{\itemsep}{0.2em}
  \item $J$ — «дифференцирование с поправкой» по каждому аргументу. Поэтому $J=0$ на $\Sm$, как только $J=0$ на $\Lh$; а на $\Lh$ это тождество Якоби в $\LL$ (и $1\in\Lh$).
  \item Итог (\go{finish}{предложение 10, теоремы 11, 12}): $\Sm$ — обобщённая алгебра Герстенхабера, $\varphi\colon\G\to\Sm$ переводит $U$ в базис $\Sm$, значит $U$ — базис $\G$. \hfill$\square$
\end{itemize}
\end{frame}

\partslide{05}{Размерность}

\begin{frame}[label=multi]{\ft{Мультилинейная часть: $n\cdot n!$}{Мультилинейная часть: n·n!}}
$X=\{1,x_1,\dots,x_n\}$; \ $\mathrm{GenGer}(n)$ — оболочка элементов $U$, где каждая из букв $1,x_1,\dots,x_n$ входит ровно один раз. \hlt{Буква $1$ тоже считается.}
\vskip0.15cm
\begin{thmbox}{Теорема 14}
\centering\Large $\dim\mathrm{GenGer}(n)=n\cdot n!$
\end{thmbox}
\vskip0.15cm
\centering\small\renewcommand{\arraystretch}{1.2}
\begin{tabular}{@{}lcccc@{}}
\toprule
$n$ & 1 & 2 & 3 & 4\\
\midrule
обобщённые алгебры Герстенхабера (и Пуассона, K17): $n\cdot n!$ & \textbf{1} & \textbf{4} & \textbf{18} & \textbf{96}\\
алгебры Пуассона и Герстенхабера, по $x_1,\dots,x_n$: $n!$ & 1 & 2 & 6 & 24\\
\bottomrule
\end{tabular}
\vskip0.25cm
\raggedright
{\small При $n=2$ — четыре элемента \go{cor13}{примера 2}. Почему $n\cdot n!$ — на следующем слайде.}
\end{frame}

\begin{frame}[label=perm]{\ft{Подсчёт: перестановки, где $1$ не на месте}{Подсчёт: перестановки, где 1 не на месте}}
\begin{itemize}\small\setlength{\itemsep}{0.25em}
\item Элемент $\mathrm{GenGer}(n)$ — разбиение $Y=\{1,x_1,\dots,x_n\}$ на блоки и мультилинейное хорошее слово на каждом блоке; квадраты $\{v,v\}$ не мультилинейны. \hlt{Блока $\{1\}$ нет}: множителя $1$ в $U$ нет.
\item Хороших слов на $k$ буквах $(k-1)!$ — как циклов на $k$ элементах. Разбиение с циклом на каждом блоке — это перестановка $Y$.
\item Убираем перестановки, где $1$ — неподвижная точка: \quad $(n+1)!-n!=n\cdot n!$.
\end{itemize}
\begin{exbox}{Пример: $n=2$, \ $Y=\{1,x_1,x_2\}$, \ $3!=6$ перестановок}
\small\renewcommand{\arraystretch}{1.2}
\begin{tabular}{@{}l@{\qquad}l@{}}
$(1\,x_1)(x_2)$ & $\{x_1,1\}\,x_2$\\
$(1\,x_2)(x_1)$ & $\{x_2,1\}\,x_1$\\
$(1\,x_1\,x_2)$, \ $(1\,x_2\,x_1)$ & $\{\{x_1,1\},x_2\}$, \ $\{\{x_2,1\},x_1\}$\\
{\color{mute}$(1)(x_1)(x_2)$, \ $(1)(x_1\,x_2)$} & {\color{mute}$1$ на месте — не считаются}\\
\end{tabular}\\[0.3em]
Блок $Y$ целиком: два цикла — два хороших слова. \ Итого $6-2=4=2\cdot2!$.
\end{exbox}
\end{frame}

\begin{frame}[label=kantor]{Откуда поправка: йордановы скобки и двойник Кантора}
\begin{defbox}{Двойник Кантора $K(A)=A\oplus Ax$}
$A$ — суперкоммутативная супералгебра со скобкой $\{\,,\}$, $Ax$ — копия $A$; умножение:
$$a\ast b=ab,\qquad a\ast bx=(ab)x,\qquad ax\ast b=(-1)^{|b|}(ab)x,\qquad ax\ast bx=(-1)^{|b|}\{a,b\}.$$
\end{defbox}
\begin{itemize}\small\setlength{\itemsep}{0.35em}
  \item Кантор (1990): для скобки Пуассона $K(A)$ — \tip{йорданова супералгебра}{суперкоммутативная алгебра с тождеством Йордана (x²y)x = x²(yx), с суперзнаками}.
  \item Кинг--МакКриммон (1992): $K(A)$ йорданова $\iff$ $\{\,,\}$ — \hlt{йорданова скобка}: Лейбниц и Якоби выполнены с поправками через $\{a,1\}$.
  \item Замена $\{a,b\}\mapsto\{a,b\}+\tfrac12\bigl(a\{b,1\}-\{a,1\}b\bigr)$ делает йорданову скобку обобщённой пуассоновой: Якоби точное, поправка только в Лейбнице.
  \item Нечётный вариант — \hlt{обобщённые алгебры Герстенхабера} (Кантарини--Кац 2010).
\end{itemize}
\end{frame}

\begin{frame}[label=po]{\ft{Примеры с $\D\neq0$: $PO(n,n+1)$}{Примеры с ненулевым D: PO(n,n+1)}}
\small
К $PO(n,n)$ добавим нечётную переменную $\tau$: $PO(n,n+1)=\F[x_1,\dots,x_n\mid\xi_1,\dots,\xi_n,\tau]$ со скобкой ($\{\,,\}_H$ — скобка Баттина, $E$ — оператор Эйлера)
$$\{f,g\}_K=\{f,g\}_H+(E-2)(f)\,\frac{\partial g}{\partial\tau}+(-1)^{|f|}\frac{\partial f}{\partial\tau}\,(E-2)(g),\qquad
E=\sum_{i=1}^n\Bigl(x_i\frac{\partial}{\partial x_i}+\xi_i\frac{\partial}{\partial\xi_i}\Bigr).$$
\begin{itemize}\setlength{\itemsep}{0.15em}
  \item $\D(g)=\{1,g\}_K=(E-2)(1)\,\partial g/\partial\tau=-2\,\partial g/\partial\tau\neq0$; \ например, $\{1,\tau\}_K=-2$, но $\{1,1\}_K=0$.
  \item Здесь $\D$ — всё-таки дифференцирование, так как $\{1,1\}=0$. В свободной алгебре будет иначе.
\end{itemize}
\normalsize
\begin{thmbox}{Теорема (Кантарини--Кац, 2010)}
Всякая простая линейно компактная обобщённая алгебра Герстенхабера калибровочно эквивалентна $PO(n,n)$ или $PO(n,n+1)$.
\end{thmbox}
\end{frame}

\begin{frame}[label=jacobi]{Якобиатор}
\begin{thmbox}{Лемма 9}
Пусть выполнены \ax{1}, \ax{2}, \ax{3}, \ax{5} (без тождества Якоби), и
$$J(a,b,c)=\{a,\{b,c\}\}-\{\{a,b\},c\}-(-1)^{\rp a\rp b}\{b,\{a,c\}\}.$$
Тогда
\begin{enumerate}
  \item[(i)] $J(a,b,c)=-(-1)^{\rp a\rp b}J(b,a,c)=-(-1)^{\rp b\rp c}J(a,c,b)$;
  \item[(ii)] $J(a,b,xy)=J(a,b,x)\,y+(-1)^{(|a|+|b|)|x|}\,x\,J(a,b,y)\;{\color{deepgreen}-\,J(a,b,1)\,xy}$.
\end{enumerate}
\end{thmbox}
\begin{itemize}\small
  \item \tip{Якобиатор}{мера нарушения тождества Якоби: J = 0 на всех тройках ⇔ тождество Якоби выполнено} — «дифференцирование с поправкой»; без поправки это классический факт для алгебр Пуассона.
  \item Доказательство: раскрыть три члена по \ax{5}, лишние члены сокращаются попарно.
  \item Общая формула, может быть полезна сама по себе.
\end{itemize}
\end{frame}

\begin{frame}[label=finish]{Тождество Якоби и завершение}

  \item \textbf{Предложение 10.} $J=0$ на $\Sm$:
  \begin{itemize}\small
    \item $J(u,v,w)=0$ при $u,v,w\in\Lh$ — это Якоби в $\LL$;
    \item по (ii) и индукции по числу множителей — для любого третьего аргумента;
    \item по (i) — для всех аргументов.
  \end{itemize}
  \item \textbf{Теорема 11.} $(\Sm,\cdot,\{\,,\}^*)$ — унитальная обобщённая алгебра Герстенхабера, порождённая $X$.

  вместе с \go{lemma5}{леммой 5}: $U$ — базис $\G$ и $\varphi$ — изоморфизм. \hfill$\square$
\end{enumerate}
